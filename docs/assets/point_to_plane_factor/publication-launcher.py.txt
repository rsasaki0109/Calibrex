DEST = Path('docs/assets/point_to_plane_factor')

WORKSPACE = str(Path.cwd())
MODULE_PATHS = {'lidar_point_to_plane': 'src/calibrex/graph/lidar_point_to_plane.py'}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def normalize(value):
    if isinstance(value, dict):
        return {normalize(k): normalize(v) for k, v in value.items()}
    if isinstance(value, list):
        return [normalize(v) for v in value]
    if isinstance(value, str):
        value = value.replace(WORKSPACE, '${WORKSPACE_ROOT}').replace(
            WORKSPACE.replace('//', '/'), '${WORKSPACE_ROOT}')
        if '${WORKSPACE_ROOT}' in value:
            value = value.replace('//', '/')
    return value


def check_worker(root, method, repeat, canonical, metadata):
    directory = root / f'{method}.repeat-{repeat:02d}'
    assert hashlib.sha256(canonical.encode()).hexdigest()
    assert sha(directory / 'result.yaml') == metadata['result_sha256']
    assert validate_file(directory / 'result.yaml').valid
    written = _normalized_result(
        load_result(directory / 'result.yaml').model_dump(mode='json'), directory.resolve())
    assert json.dumps(written, sort_keys=True, separators=(',', ':'), allow_nan=False) == canonical
    for filename, digest in metadata['artifacts'].items():
        assert sha(directory / filename) == digest


def publish(root, target, commit):
    target.mkdir(parents=True, exist_ok=True)
    raw = BenchmarkDefinition.model_validate_json((root / 'pipeline.definition.json').read_bytes())
    recorded_commit = subprocess.check_output(
        ['git', 'rev-parse', raw.methods[1].source_commit + '^{commit}'], text=True).strip()
    assert len(raw.trials) == 10 and recorded_commit == commit
    anchor = raw.trials[0].provenance.notes[-1]
    for trial in raw.trials:
        assert trial.status == 'success' and trial.metrics['complete_result_matches'] == 1
        canonical = trial.provenance.notes[-1]
        assert canonical == anchor
        assert hashlib.sha256(canonical.encode()).hexdigest() == trial.provenance.output_sha256
        metadata = json.loads(trial.provenance.notes[0])
        check_worker(root, trial.method_id, metadata['repeat'], canonical, metadata)
    bindings = {}
    for path in sorted(root.glob('baseline.*.py.txt')):
        name = path.name.removeprefix('baseline.').removesuffix('.py.txt')
        source = MODULE_PATHS[name]
        data = subprocess.check_output(['git', 'show', raw.methods[0].source_commit + ':' + source])
        assert path.read_bytes() == data
        archived = target / path.name
        archived.write_bytes(data)
        bindings['published:' + archived.name] = sha(archived)
        bindings['baseline:' + source] = sha(archived)
    for source in raw.provenance.source_artifacts:
        if not source.startswith(('src/', 'tools/')):
            continue
        archived = target / ('measured.' + Path(source).name + '.txt')
        archived.write_bytes(subprocess.check_output(['git', 'show', commit + ':' + source]))
        bindings['published:' + archived.name] = sha(archived)
        bindings['committed:' + source] = sha(archived)
    launcher = target / 'publication-launcher.py.txt'
    launcher.write_text(normalize(Path(__file__).read_text(encoding='utf-8')).rstrip() + '/n', encoding='utf-8', newline='/n')
    bindings['published:' + launcher.name] = sha(launcher)
    payload = normalize(raw.model_dump(mode='json'))
    payload['methods'][1]['source_commit'] = commit
    for trial, original in zip(payload['trials'], raw.trials, strict=True):
        trial['provenance']['notes'][0] = json.dumps(
            normalize(json.loads(original.provenance.notes[0])), sort_keys=True)
        trial['provenance']['notes'][-1] = original.provenance.notes[-1]
    payload['provenance']['source_artifacts'].update(bindings)
    payload['provenance']['source_artifacts']['original:definition'] = sha(root / 'pipeline.definition.json')
    payload['provenance']['source_artifacts']['original:aggregate'] = sha(root / 'pipeline.benchmark.json')
    payload['limitations'].append(
        'Publication normalizes workspace prefixes in commands/metadata only. Complete canonical '
        'result strings and original input/config/output/artifact hashes remain unchanged. '
        'Archived code is exact Git bytes; working-byte and committed-byte hashes are separate. '
        'Recorded short candidate commit IDs are resolved to their full Git commit; canonical '
        'result strings retain the original short run.git_commit.')
    payload['limitations'].append(
        'Workers call the public calibration Python API; CLI dispatch is excluded. Worker '
        'measurement JSON writes are included in wall. No timings or outliers are discarded.')
    definition = BenchmarkDefinition.model_validate(payload)
    definition.save(target / 'pipeline.definition.json')
    aggregate_benchmark_definition(definition).save(target / 'pipeline.benchmark.json')
    return definition

## Numerical contract and quality

At 64 or more observations, cache read-only observation arrays within the factor
and compute residuals in chunks of at most 4,096 observations. Replay the original
Hamilton products and **all four composed-rotation normalizations**, including
the two in point rotation. Preserve the translation composition, ordered dot
products, square-root weights, residual order and numerical finite differences.
Hessian/gradient accumulation, robust weighting and optimizer logic are unchanged.
Small populations and nonfinite input/transform values retain the scalar path.
Replacing the observation tuple refreshes the cache; changing the initial transform
is observed on the next call. Retained arrays cost 17 float64 values per observation
(136 bytes), plus array objects; temporaries are bounded by the chunk size.

Tests compare residual bits for static and per-point moving poses, chunk boundaries,
weights, signed zeros, nonfinite and overflow cases; they compare complete numerical
linearization, observability and solver history against the original scalar formula.

The optimizer still reports `converged`, but observability rank remains **3/6**
and overall quality remains **warn**. Faster execution does not establish full
six-DoF observability or absolute calibration accuracy. This single public capture
