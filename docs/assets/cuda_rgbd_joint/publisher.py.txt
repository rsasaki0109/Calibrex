import gzip

OUT=ROOT/'outputs/cuda-rgbd-joint-pairs-v3'

TARGET=ROOT/'docs/assets/cuda_rgbd_joint'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()

def write(path,value):
    path.write_bytes((json.dumps(value,indent=2,allow_nan=False)+'\n').encode())

def schema(value):
    if value is None: return {'type':'null'}
    if isinstance(value,bool): return {'type':'boolean'}
    if isinstance(value,int): return {'type':'integer'}
    if isinstance(value,float): return {'type':'number'}
    if isinstance(value,str): return {'type':'string'}
    if isinstance(value,list):
        choices={json.dumps(schema(v),sort_keys=True) for v in value}
        items={} if not choices else json.loads(next(iter(choices))) if len(choices)==1 else {'anyOf':[json.loads(v) for v in sorted(choices)]}
        return {'type':'array','minItems':len(value),'maxItems':len(value),'items':items}
    return {'type':'object','additionalProperties':False,'required':list(value),'properties':{k:schema(v) for k,v in value.items()}}

def source_capsule(source,name):
    shutil.copyfile(source/'manifest.json',TARGET/f'sources-{name}-manifest.json')
    with tarfile.open(TARGET/f'sources-{name}.tar.gz','w:gz') as bundle:
        for path in sorted((source/'src').rglob('*.py')):
            bundle.add(path,arcname=path.relative_to(source).as_posix())

def records(root,prefix=''):
    for path in sorted(root.rglob('*')):
        if not path.is_file(): continue
        parts=path.relative_to(root).parts
        if any('cache' in part.lower() or part in ('source','sources-current','sources-prior','shared-output') for part in parts[:-1]): continue
        if path.suffix not in ('.json','.yaml','.txt','.gz','.log'): continue
        yield path,prefix+path.relative_to(root).as_posix()
