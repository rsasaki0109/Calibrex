"""Bind completed CPU/GPU and CPU-only checks to the exact measured runtime."""
import hashlib
import json
from pathlib import Path
import re
import shutil

ROOT=Path(__file__).resolve().parent
TARGET=ROOT/'docs/assets/cuda_rgbd_joint'
VALIDATION=ROOT/'outputs/cuda-rgbd-joint-validation-v1'
RUNTIME='e3f39bbfc7fb07df730d9f7bebf41e3fa70c14c0'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()

def write(path,value):
    path.write_bytes((json.dumps(value,indent=2)+'\n').encode())

manifest=json.loads((TARGET/'sources-current-manifest.json').read_bytes())
assert len(manifest)==392
assert sha(ROOT/'tools/audit_cuda_rgbd_joint.py')==sha(TARGET/'auditor.py.txt')
counts={}
logs={}
for mode in ('core','archive'):
    for platform in ('windows','linux'):
        name=f'{mode}-{platform}'
        env=json.loads((VALIDATION/(name+'-environment.json')).read_bytes())
        assert env['exit_status']==0 and env['provenance']['sources']==manifest
        assert env['provenance']['git_commit']==RUNTIME
        assert all(manifest[v['path']]==v['sha256'] for v in env['imports'].values())
        assert all(sha(ROOT/path)==value for path,value in env['test_files_sha256'].items())
        text=(VALIDATION/(name+'.log')).read_text(encoding='utf-8')
        counts[name]=int(re.findall(r'(\d+) passed',text)[-1])
        assert counts[name]==(9 if mode=='archive' else 539 if platform=='windows' else 505)
        if mode=='core' and platform=='linux': assert '1 skipped' in text
        for suffix in ('.log','-environment.json'):
            shutil.copyfile(VALIDATION/(name+suffix),TARGET/(name+suffix))
        logs[name+'.log']=sha(TARGET/(name+'.log'))
for platform in ('linux','win32'):
    source=(ROOT/'outputs/cuda-rgbd-joint-development-v1/mypy-linux.log'
            if platform=='linux' else VALIDATION/'mypy-win32.log')
    assert 'Success: no issues found in 392 source files' in source.read_text()
    shutil.copyfile(source,TARGET/f'mypy-{platform}.log')
    logs[f'mypy-{platform}.log']=sha(TARGET/f'mypy-{platform}.log')
parity=json.loads((VALIDATION/'schema-parity.json').read_bytes())
assert parity['schema_exports_and_ledger']==130 and parity['all_exports_byte_identical']
shutil.copyfile(VALIDATION/'schema-parity.json',TARGET/'schema-parity.json')
for source,name in [
    ('.tmp-calibrex-cuda-rgbd-frozen-tests-v1.py','frozen-test-runner.py.txt'),
    ('.tmp-calibrex-cuda-rgbd-linux-v1.sh','linux-test-runner.sh.txt'),
    ('.tmp-calibrex-cuda-rgbd-archive-linux-v1.sh','linux-archive-test-runner.sh.txt'),
    ('.tmp-calibrex-cuda-rgbd-schema-parity-v1.py','schema-parity.py.txt'),
    ('tests/unit/test_rgbd_execution.py','test_rgbd_execution.py.txt'),
    ('tests/unit/test_cuda_rgbd_joint_adapter.py','test_cuda_rgbd_joint_adapter.py.txt'),
    ('tests/unit/test_cuda_rgbd_joint_archive.py','test_cuda_rgbd_joint_archive.py.txt'),
]: shutil.copyfile(ROOT/source,TARGET/name)
shutil.copyfile(Path(__file__),TARGET/'validation-finalizer.py.txt')
record={'test_pass_counts':counts,'linux_optional_cuda_test_module_skipped':1,
        'real_device_cases_windows':34,'mypy_source_modules':{'linux':392,'win32':392},
        'public_schema_exports_and_ledger_byte_identical':130,
        'complete_optimizer_results_and_array_traces_exact':6,
        'provenance':{'runtime_commit':RUNTIME,'source_manifest_sha256':sha(TARGET/'sources-current-manifest.json'),
                      'measurement_provenance_sha256':sha(TARGET/'provenance.json'),
                      'generator_sha256':sha(Path(__file__)),
                      'typecheck_requirements_sha256':sha(ROOT/'tools/typecheck-requirements.txt')},
        'log_sha256':logs}
for name,key,marker in [('ruff.log','ruff_passed','All checks passed!'),('mkdocs.log','strict_mkdocs_passed','Documentation built in ')]:
    if (VALIDATION/name).exists():
        assert marker in (VALIDATION/name).read_text(encoding='utf-8',errors='replace')
        shutil.copyfile(VALIDATION/name,TARGET/name)
        record[key]=True
        logs[name]=sha(TARGET/name)
write(TARGET/'validation.json',record)
index=json.loads((TARGET/'index.json').read_bytes())
index['files']={p.relative_to(TARGET).as_posix():{'sha256':sha(p),'size_bytes':p.stat().st_size}
                for p in sorted(TARGET.rglob('*')) if p.is_file() and p.name!='index.json'}
write(TARGET/'index.json',index)
print('Bound all completed checks to',RUNTIME,';',len(index['files'])+1,'archive files')
