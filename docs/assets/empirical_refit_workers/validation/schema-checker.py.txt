"""Untimed schema and complete runtime comparison against both frozen revisions."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

ROOT = Path(__file__).resolve().parent
RUN = ROOT / 'outputs/empirical-refits-formal-v2'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
protocol = json.loads((RUN / 'protocol.json').read_bytes())
for variant in ('prior', 'current'):
    source = RUN / ('sources-' + variant)
    manifest = json.loads((RUN / ('sources-' + variant + '-manifest.json')).read_bytes())
    assert manifest == {p.relative_to(source / 'src').as_posix(): sha(p)
                        for p in (source / 'src').rglob('*.py')}
    target = RUN / ('schemas-' + variant)
    env = os.environ.copy()
    env.update(PYTHONPATH=str(source / 'src'), PYTHONDONTWRITEBYTECODE='1')
    code = "from calibrex.cli.main import main; import sys; raise SystemExit(main(['schema','all','--output-dir',sys.argv[1]]))"
    done = subprocess.run([sys.executable, '-c', code, str(target)], env=env, cwd=source,
                          capture_output=True, check=True)
    (RUN / ('schema-' + variant + '-stdout.txt')).write_bytes(done.stdout)
    (RUN / ('schema-' + variant + '-stderr.txt')).write_bytes(done.stderr)
    assert manifest == {p.relative_to(source / 'src').as_posix(): sha(p)
                        for p in (source / 'src').rglob('*.py')}
names = {p.name for p in (RUN / 'schemas-prior').glob('*.json')}
assert names == {p.name for p in (RUN / 'schemas-current').glob('*.json')}
exports = {n: {v: sha(RUN / ('schemas-' + v) / n) for v in ('prior', 'current')}
           for n in sorted(names)}
assert len(exports) == 130
assert all(v['prior'] == v['current'] for v in exports.values())
source = RUN / 'sources-current/src'
live = ROOT / 'src'
measured_files = {p.relative_to(source).as_posix(): p for p in source.rglob('*.py')}
live_files = {p.relative_to(live).as_posix(): p for p in live.rglob('*.py')}
assert measured_files.keys() == live_files.keys()
norm = lambda p: hashlib.sha256(p.read_bytes().replace(b'\r\n', b'\n')).hexdigest()
assert all(norm(live_files[n]) == norm(measured_files[n]) for n in live_files)
receipt = {
    'exports': exports,
    'runtime_sources': len(live_files),
    'runtime_matches_measured_candidate': True,
    'newline_normalization': 'CRLF to LF in both operands',
    'provenance': {
        'source_manifests': protocol['provenance']['manifests'],
        'protocol_sha256': sha(RUN / 'protocol.json'),
        'checker_sha256': sha(Path(__file__)),
    },
}
(RUN / 'schema-parity.json').write_bytes((json.dumps(receipt, indent=2) + '\n').encode())
print(f'All {len(exports)} frozen schema/ledger exports are byte-identical; all {len(live_files)} runtime sources match the measured candidate.')
