"""Preserve measured candidates, including negative and invalid instrumented screens."""
import gzip
import hashlib
import json
from pathlib import Path
import shutil
import tarfile

ROOT=Path(__file__).resolve().parent
OUT=ROOT/'outputs/tum-diagnostic-production-v1'
TARGET=ROOT/'docs/assets/tum_diagnostic_execution'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()

def write(path,value):
    path.write_bytes((json.dumps(value,indent=2,allow_nan=False)+'\n').encode())

def schema(value):
    if value is None:return {'type':'null'}
    if isinstance(value,bool):return {'type':'boolean'}
    if isinstance(value,int):return {'type':'integer'}
    if isinstance(value,float):return {'type':'number'}
    if isinstance(value,str):return {'type':'string'}
    if isinstance(value,list):
        choices={json.dumps(schema(item),sort_keys=True) for item in value}
        items={} if not choices else json.loads(next(iter(choices))) if len(choices)==1 else {'anyOf':[json.loads(item) for item in sorted(choices)]}
        return {'type':'array','minItems':len(value),'maxItems':len(value),'items':items}
    return {'type':'object','additionalProperties':False,'required':list(value),'properties':{key:schema(item) for key,item in value.items()}}

def capsule(source,label):
    shutil.copyfile(source/'manifest.json',TARGET/f'sources-{label}-manifest.json')
    with tarfile.open(TARGET/f'sources-{label}.tar.gz','w:gz') as bundle:
        for p in sorted((source/'src').rglob('*.py')):
            bundle.add(p,arcname=p.relative_to(source).as_posix())

summary=json.loads((OUT/'summary.json').read_bytes())
assert len(summary['attempts'])==6 and summary['complete_normalized_results_exact']
TARGET.mkdir(exist_ok=True)
(TARGET/'.gitattributes').write_bytes(b'* -text whitespace=cr-at-eol\n')
for name in ('config.yaml','provenance.json','attempts.json','summary.json','launcher.py.txt'):
    shutil.copyfile(OUT/name,TARGET/name)
for source,name in [('docs/assets/tum_window_processes/inputs.json','inputs.json'),
                    ('tools/audit_tum_diagnostic_execution.py','auditor.py.txt')]:
    shutil.copyfile(ROOT/source,TARGET/name)
shutil.copyfile(Path(__file__),TARGET/'publisher.py.txt')
shutil.copyfile(ROOT/'.tmp-calibrex-publish-tdg-v1.py',TARGET/'publisher-incomplete-v1.py.txt')
for variant in ('prior','current'):
    capsule(OUT/('sources-'+variant),variant)
capsule(ROOT/'outputs/tum-diagnostics-pairs-v1/current/sources-current','batched')
capsule(ROOT/'outputs/tum-probe-process-screen-v1/sources-current','probes')
records={'provenance':[json.loads((OUT/'provenance.json').read_bytes())],
         'attempts':[json.loads((OUT/'attempts.json').read_bytes())],
         'summary':[summary],'measurement':[],'result':[]}
for attempt in summary['attempts']:
    folder=TARGET/attempt['folder']
    folder.mkdir()
    for name in ('measurement.json','result.json.gz','stdout.log','stderr.log'):
        shutil.copyfile(OUT/folder.name/name,folder/name)
    records['measurement'].append(json.loads((folder/'measurement.json').read_bytes()))
    records['result'].append(json.loads(gzip.decompress((folder/'result.json.gz').read_bytes())))
schemas={}
for kind,values in records.items():
    choices={json.dumps(schema(value),sort_keys=True) for value in values}
    schemas[kind]=json.loads(next(iter(choices))) if len(choices)==1 else {'anyOf':[json.loads(item) for item in sorted(choices)]}
    schemas[kind]['$schema']='https://json-schema.org/draft/2020-12/schema'
schemas['calibration-result']=json.loads((ROOT/'schemas/result.schema.json').read_bytes())
write(TARGET/'record-schemas.json',schemas)
traces=ROOT/'outputs/tum-diagnostic-traces-v1'
trace_summary=json.loads((traces/'summary.json').read_bytes())
assert trace_summary['fixed_population_probe_events_exact']==492
with tarfile.open(TARGET/'supplementary-traces.tar.gz','w:gz') as bundle:
    for name in ('provenance.json','attempts.json','summary.json','launcher.py.txt'):
        bundle.add(traces/name,arcname=name)
    for variant in ('prior','current'):
        source=traces/variant/'sources-current'
        assert json.loads((source/'manifest.json').read_bytes())==json.loads((OUT/('sources-'+variant)/'manifest.json').read_bytes())
        for name in ('config.yaml','protocol.json'):
            bundle.add(traces/variant/name,arcname=variant+'/'+name)
        bundle.add(source/'manifest.json',arcname=variant+'/source-manifest.json')
    for attempt in trace_summary['attempts']:
        for p in sorted((traces/attempt['folder']).rglob('*')):
            if p.is_file():bundle.add(p,arcname=p.relative_to(traces).as_posix())
    bundle.add(ROOT/'tools/benchmark_tum_window_processes.py',arcname='trace-worker.py.txt')
assert sha(ROOT/'tools/benchmark_tum_window_processes.py')==json.loads((traces/'provenance.json').read_bytes())['trace_worker_sha256']
development={
 'baseline-profile':'outputs/tum-small-process-profile-v1',
 'batched-traced-screen':'outputs/tum-diagnostics-pairs-v1',
 'probe-process-screen':'outputs/tum-probe-process-screen-v1',
 'stage-profile-v1-failed-flush':'outputs/tum-diagnostic-light-profile-v1',
 'stage-profile-v2-disables-native-map':'outputs/tum-diagnostic-light-profile-v2',
 'stage-profile-v3':'outputs/tum-diagnostic-light-profile-v3',
}
excluded={'source','sources-prior','sources-current','shared-output','run','.git','__pycache__'}
dev_index={}
with tarfile.open(TARGET/'development-screens.tar.gz','w:gz') as bundle:
    for label,relative in development.items():
        files=[]
        folder=ROOT/relative
        for p in sorted(folder.rglob('*')):
            name=p.relative_to(folder)
            if not p.is_file() or any(part in excluded for part in name.parts):continue
            bundle.add(p,arcname=label+'/'+name.as_posix())
            files.append({'name':name.as_posix(),'sha256':sha(p),'size_bytes':p.stat().st_size})
        dev_index[label]={'files':files,'scope':'Exploratory or invalid instrumented screen. Timings are not production latency evidence.'}
    for name in ('.tmp-calibrex-tum-small-process-profile-v1.py','.tmp-calibrex-tum-small-profile-compare-v1.py'):
        bundle.add(ROOT/name,arcname='baseline-profile/'+name.removeprefix('.tmp-calibrex-')+'.txt')
write(TARGET/'development-index.json',{'records':dev_index,'provenance':{'publisher_sha256':sha(Path(__file__))}})
for name in ('test_owned_voxel_geometry','test_small_captured_plane_queries','test_native_diagnostic_scoring'):
    shutil.copyfile(ROOT/f'tests/unit/{name}.py',TARGET/(name+'.py.txt'))
shutil.copyfile(ROOT/'tools/typecheck-requirements.txt',TARGET/'typecheck-requirements.txt')
write(TARGET/'index.json',{'files':{p.relative_to(TARGET).as_posix():{'sha256':sha(p),'size_bytes':p.stat().st_size} for p in sorted(TARGET.rglob('*')) if p.is_file() and p.name!='index.json'},'provenance':{'production_provenance_sha256':sha(TARGET/'provenance.json'),'publisher_sha256':sha(Path(__file__))}})
print('Preserved original production and supplementary records, without promoting the candidates.')
