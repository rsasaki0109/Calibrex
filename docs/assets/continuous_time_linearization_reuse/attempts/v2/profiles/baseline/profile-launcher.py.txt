def observe(owner, name):
    native = getattr(owner, name)
    label = (owner.__name__ if hasattr(owner,'__name__') else str(owner))+'.'+name
    def observed(*args, **kwargs):
        start = time.perf_counter()
        try:
            return native(*args, **kwargs)
        finally:
            elapsed = time.perf_counter()-start
            with lock:
                row = counts.setdefault(label, {'calls':0,'sum_seconds':0.0})
                row['calls'] += 1
                row['sum_seconds'] += elapsed
    setattr(owner, name, observed)

native_gpu_linearize = CudaRigLinearizer.linearize
def count_duplicate(self, arrays, transforms, steps):
    import numpy as np
    key = (id(arrays), np.asarray([(*t.translation_m, *t.rotation_quat_xyzw) for t in transforms], dtype=np.float64).tobytes(), np.asarray(steps, dtype=np.float64).tobytes())
    seen = getattr(self, '_profile_seen', set())
    duplicate = key in seen
    seen.add(key)
    self._profile_seen = seen
    start = time.perf_counter()
    result = native_gpu_linearize(self, arrays, transforms, steps)
    elapsed = time.perf_counter()-start
    with lock:
        row = counts.setdefault('gpu_repeated_identical_payload' if duplicate else 'gpu_unique_payload', {'calls':0, 'rows':0, 'sum_seconds':0.0})
        row['calls'] += 1
        row['rows'] += len(arrays.points)
        row['sum_seconds'] += elapsed
    return result
CudaRigLinearizer.linearize = count_duplicate
