timers = {}
lock = Lock()

def observed_gpu(name, label):
    native = getattr(CudaRigLinearizer, name)
    def observed(self, arrays, *args):
        start = time.perf_counter()
        before = self.linearization_calls
        hits = self.linearization_cache_hits
        result = native(self, arrays, *args)
        with lock:
            launched = self.linearization_calls-before if name=='linearize' else 1
            counts[label+'_calls'] += launched
            counts[label+'_rows'] += len(arrays.points)*launched
            if name=='linearize':
                counts['gpu_linearization_requests'] += 1
                counts['gpu_linearization_cache_hits'] += self.linearization_cache_hits-hits
                counts['max_per_context_cache_bytes'] = max(counts['max_per_context_cache_bytes'], self.linearization_cache_bytes)
            timers[label] = timers.get(label, 0.0)+time.perf_counter()-start
        return result
    setattr(CudaRigLinearizer, name, observed)

capture = CudaRigLinearizer._capture
def observed_capture(self, arrays):
    changed = self._arrays is not arrays
    result = capture(self, arrays)
    if changed:
        with lock:
            counts['gpu_payload_uploads'] += 1
            counts['gpu_payload_bytes'] += sum(v.nbytes for v in (
                arrays.points, arrays.plane_points, arrays.normals, arrays.poses, arrays.sqrt_weights))
    return result
CudaRigLinearizer._capture = observed_capture

    started = time.perf_counter()
    before = self.residual_calls
    result = native_captured(self, arrays, transform)

        if result is not None:

            counts['gpu_summary_cache_hits'] += int(launched == 0)
        timers['gpu_captured_summary'] = timers.get('gpu_captured_summary',0.0)+time.perf_counter()-started
    return result

                  ('numba_rig_residuals','cpu_residual_calls'),

    def wrap(native, key):
        def observed(*args, **kwargs):
            started = time.perf_counter()
            result = native(*args, **kwargs)
            with lock:
                counts[key] += 1
                if key == 'cpu_normal_calls':
                    caller = sys._getframe(1)
                    if caller.f_code.co_name == '_compiled_system' and not caller.f_locals['_include_residuals']:

                        counts['discarded_residual_rows_avoided'] += len(args[1])
                timers[key] = timers.get(key, 0.0)+time.perf_counter()-started
            return result
        return observed
    setattr(a, name, wrap(getattr(a, name), key))

repair = AdaptivePlaneQueryContext._positions_from_neighbors
def observed_repair(self, points, distances, indexes, gate):
    first = distances[:,0]
    margin = 32.0*np.finfo(np.float64).eps
    radii = np.nextafter(first*(1.0+margin), np.inf)
    lower = np.nextafter(first*(1.0-margin), -np.inf)
    fast = np.isfinite(first) & (first>=1e-150) & (distances[:,1]>radii) & ((radii<=gate)|(lower>gate))
    with lock:
        counts['cpu_adaptive_repaired_rows'] += int(np.count_nonzero(~fast))
    return repair(self, points, distances, indexes, gate)
AdaptivePlaneQueryContext._positions_from_neighbors = observed_repair
