"""Audit both frozen cohorts and publish exact source/complete-result evidence."""
from __future__ import annotations

import hashlib
import importlib.util
import json
import subprocess
import sys
from pathlib import Path

from calibrex.core.benchmark import BenchmarkDefinition, aggregate_benchmark_definition
from calibrex.core.continuous_time_lidar_artifacts import ContinuousTimeLidarPairArtifact
from calibrex.core.io import read_mapping

spec = importlib.util.spec_from_file_location("b", "tools/benchmark_adaptive_plane_search.py")
b = importlib.util.module_from_spec(spec); spec.loader.exec_module(b)
destination = Path("docs/assets/continuous_time_query_reuse")
destination.mkdir(parents=True, exist_ok=True)
cohorts = [
    ("bounded", Path("outputs/public-glim-query-reuse-bounded-v1"), 1200, 1200, 894, 876, 62400),
    ("default_budget", Path("outputs/public-glim-query-reuse-default-v1"),
     6000, 3200, 4773, 2526, 184000),
]
candidate = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
baseline = "3f1e1f412d5d373493395e53b2636e079184e457"
rewrite = {}
archive_hashes = {}
for path in b.SOURCES:
    name = "measured." + Path(path).name + ".txt"
    contents = subprocess.check_output(["git", "show", f"{candidate}:{path}"])
    assert Path(path).read_bytes().replace(b"\r\n", b"\n") == contents
    (destination / name).write_bytes(contents)
    rewrite[path] = name
    archive_hashes[name] = b._digest(destination / name)
for module in b.RUNTIME_MODULES:
    path = "src/" + module.replace(".", "/") + ".py"
    name = "baseline." + module.rsplit(".", 1)[1] + ".py.txt"
    contents = subprocess.check_output(["git", "show", f"{baseline}:{path}"])
    (destination / name).write_bytes(contents)
    archive_hashes[name] = b._digest(destination / name)
    for _label, raw, *_counts in cohorts:
        assert (raw / name).read_bytes() == contents
        rewrite[(raw / name).resolve().as_posix()] = name
        rewrite[str((raw / name).resolve())] = name

workspace = Path.cwd()
def portable(value):
    if isinstance(value, dict):
        return {portable(key): portable(item) for key, item in value.items()}
    if isinstance(value, list):
        return [portable(item) for item in value]
    if isinstance(value, str):
        if value in rewrite:
            return rewrite[value]
        return (value.replace(sys.executable, "<PYTHON>")
                .replace(str(workspace), "<WORKSPACE>")
                .replace(workspace.as_posix(), "<WORKSPACE>").replace("\\", "/"))
    return value

originals = {}
for label, raw, train, holdout, matched_train, matched_holdout, query_count in cohorts:
    definition = BenchmarkDefinition.model_validate(read_mapping(raw / "pipeline.definition.json"))
    assert len(definition.trials) == 10
    assert {method.source_commit for method in definition.methods} == {candidate, baseline}
    original_bindings = definition.provenance.source_artifacts
    for path in b.SOURCES:
        assert original_bindings[path] == b._digest(Path(path))
    anchor = None
    for trial in definition.trials:
        folder = raw / f"{trial.method_id}.{trial.split_id}"
        measurement = json.loads((folder / "measurement.json").read_bytes())
        saved = ContinuousTimeLidarPairArtifact.model_validate(read_mapping(folder / "result.yaml"))
        canonical = b._canonical(saved.model_dump(mode="json", exclude_none=True))
        assert canonical == b._canonical(measurement["result"]) == trial.provenance.notes[-1]
        assert b._digest(folder / "result.yaml") == measurement["result_sha256"]
        assert hashlib.sha256(canonical.encode()).hexdigest() == trial.provenance.output_sha256
        assert trial.status == "success" and trial.metrics["complete_result_matches"] == 1.0
        assert anchor is None or anchor == canonical
        anchor = canonical
        assert saved.status == "converged" and saved.observability.rank == 6
        assert saved.train_correspondence_count == matched_train
        assert saved.holdout_correspondence_count == matched_holdout
        assert measurement["adaptive_query_calls"] == query_count
        assert measurement["sampled_split"]["train_count"] == train
        assert measurement["sampled_split"]["holdout_count"] == holdout
        assert measurement["sampled_split"]["train_ids_sha256"] != (
            measurement["sampled_split"]["holdout_ids_sha256"]
        )
    payload = definition.model_dump(mode="json", exclude_none=True)
    for trial in payload["trials"]:
        notes = trial["provenance"]["notes"]
        notes[0] = json.dumps(portable(json.loads(notes[0])), sort_keys=True)
        assert b._canonical(portable(json.loads(notes[-1]))) == notes[-1]
    payload = portable(payload)
    for name, digest in archive_hashes.items():
        payload["provenance"]["source_artifacts"][name] = digest
    published = BenchmarkDefinition.model_validate(payload)
    published.save(destination / (label + ".definition.json"))
    aggregate_benchmark_definition(published).save(destination / (label + ".benchmark.json"))
    originals[label] = {
        "measurement_sha256": {
            filename: b._digest(raw / filename)
            for filename in ["pipeline.definition.json", "pipeline.benchmark.json"]
        },
        "source_bindings": {
            key.replace(str(workspace), "<WORKSPACE>").replace(
                workspace.as_posix(), "<WORKSPACE>",
            ).replace("\\", "/"): digest for key, digest in original_bindings.items()
        },
    }

lineage = Path("data/public/glim_versatile/conversion-provenance.json")
(destination / "conversion-provenance.json").write_bytes(lineage.read_bytes())
launcher = destination / "publication-launcher.py.txt"
launcher.write_bytes(Path(__file__).read_bytes().replace(b"\r\n", b"\n"))
index = {
    "provenance": {
        "generator": "publication-launcher.py.txt",
        "candidate_source_commit": candidate, "baseline_source_commit": baseline,
        "command": ["<PYTHON>", "publication-launcher.py.txt"],
    },
    "original_cohorts": originals,
    "files": {
        path.name: b._digest(path) for path in sorted(destination.iterdir())
        if path.is_file() and path.name not in ("README.md", "publication-index.json")
    },
}
(destination / "publication-index.json").write_text(
    json.dumps(index, indent=2) + "\n", encoding="utf-8", newline="\n",
)
print("All 20 actual saved artifacts, full canonical payloads, hashes and archives verified.")
