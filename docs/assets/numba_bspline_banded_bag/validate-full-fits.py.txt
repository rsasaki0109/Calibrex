from pathlib import Path
from types import ModuleType
import hashlib
import json
import sys


def validate(root: Path) -> dict[str, int]:
    protocol = json.loads((root / 'protocol.json').read_bytes())
    report = json.loads((root / 'full-fit-validation.json').read_bytes())
    helper = root / 'fit-quality.py.txt'
    assert hashlib.sha256(helper.read_bytes()).hexdigest() == protocol['fit_options']['quality_helper_sha256']
    assert report['provenance']['protocol_sha256'] == hashlib.sha256((root / 'protocol.json').read_bytes()).hexdigest()
    assert report['provenance']['validation_helper_sha256'] == hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
    module = ModuleType('_bound_banded_fit_quality')
    module.__file__ = str(helper)
    exec(compile(helper.read_bytes(), str(helper), 'exec'), module.__dict__)
    records = []
    for variant in ('prior', 'current'):
        path = root / f'00-{variant}' / 'measurement.json'
        assert hashlib.sha256(path.read_bytes()).hexdigest() == report['provenance']['measurements_sha256'][variant]
        records.append(json.loads(path.read_bytes()))
    checks = [module._compare(left, right) for left, right in zip(records[0]['rounds'], records[1]['rounds'], strict=True)]
    assert len(checks) == 4 and all(check['passed'] for check in checks)
    assert report['all_four_fits_passed'] and report['full_fit_checks'] == checks
    return {'verified_full_fits': len(checks)}


if __name__ == '__main__':
    print(json.dumps(validate(Path(sys.argv[1]) if len(sys.argv) > 1 else Path(__file__).parent)))
