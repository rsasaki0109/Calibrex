from pathlib import Path
import hashlib
import json
import shutil
import tarfile
import subprocess
import argparse

parser=argparse.ArgumentParser()
parser.add_argument('--source',default='outputs/numba-banded-public-bag-quality-v1')
parser.add_argument('--output',default='docs/assets/numba_bspline_banded_bag')
args=parser.parse_args()
source=Path(args.source); target=Path(args.output)
summary = json.loads((source / 'summary.json').read_bytes())
protocol = json.loads((source / 'protocol.json').read_bytes())
attempts = json.loads((source / 'attempts.json').read_bytes())
assert summary['quality_passed'] and len(attempts) == 2

target.mkdir(parents=True, exist_ok=True)
for name in ('protocol.json', 'attempts.json', 'summary.json'):
    shutil.copyfile(source / name, target / name)
for attempt in attempts:
    folder = target / attempt['worker']
    folder.mkdir(exist_ok=True)
    for name in ('result.json', 'measurement.json', 'worker.log'):
        shutil.copyfile(source / attempt['worker'] / name, folder / name)
for variant in ('prior', 'current'):
    root = source / f'sources-{variant}'
    shutil.copyfile(root / 'manifest.json', target / f'sources-{variant}-manifest.json')
    with tarfile.open(target / f'sources-{variant}.tar.gz', 'w:gz') as archive:
        archive.add('LICENSE', arcname='LICENSE', recursive=False)
        for path in sorted((root / 'src').rglob('*.py')):
            archive.add(path, arcname=path.relative_to(root).as_posix(), recursive=False)
for origin, name in (
    ('tools/benchmark_bspline_lidar_imu.py', 'benchmark.py.txt'),
    ('tools/benchmark_bspline_planes.py', 'fit-quality.py.txt'),
    ('tools/audit_bspline_lidar_imu_benchmark.py', 'audit.py.txt'),
    ('tools/schemas/bspline_lidar_imu_comparison.schema.json', 'comparison.schema.json'),
    (__file__, 'publisher.py.txt'),
):
    if name == 'benchmark.py.txt':
        expected=protocol['launcher_sha256']
    elif name == 'fit-quality.py.txt':
        expected=protocol['fit_options']['quality_helper_sha256']
    else:
        expected=None
    raw=Path(origin).read_bytes()
    if expected and hashlib.sha256(raw).hexdigest()!=expected:
        git_raw=subprocess.check_output(['git','show',protocol['current_commit']+':'+origin])
        variants=(git_raw,git_raw.replace(b'\n',b'\r\n'))
        raw=next(value for value in variants if hashlib.sha256(value).hexdigest()==expected)
    (target/name).write_bytes(raw)

shutil.copyfile(source/'full-fit-validation.json',target/'full-fit-validation.json')
shutil.copyfile('.tmp-calibrex-validate-banded-full-fits-v1.py',target/'validate-full-fits.py.txt')
readme = f"""# Structured spline systems: complete public bag regression

One fixed-order fresh-process pair compares compiled runtime v0.2
(`{protocol['baseline_commit']}`) and v0.3 (`{protocol['current_commit']}`).
Both read the original public GLIM Livox bag's first six seconds, using CPU
odometry, the same default geometry, four map rounds and four fit workers.
Each worker starts with an empty private Numba cache. The complete worker
includes imports, decoding, all fits/covariance, noise updates, holdout,
32 signed controls, hashes, saves and verified frozen imports.

| Scope | Prior compiled | Structured compiled | Observed ratio |
| --- | ---: | ---: | ---: |
| Complete worker | 54.809 s | 67.962 s | 0.806x |
| All four fits, including JIT and covariance | 13.826 s | 12.719 s | 1.087x |
| First fit | 5.884 s | 8.963 s | 0.657x |
| Subsequent three fits combined | 7.942 s | 3.756 s | 2.115x |

These exploratory timings do not establish whole-pipeline acceleration.
Host resources were unreserved. Unchanged odometry also varies from 24.242 s
to 36.494 s; first-use compilation is heavier in the new path. All raw times
and the negative whole-worker result are retained. No density, return,
iteration, validation or numerical-grid budget is changed.

Complete decisions and numerical evidence pass the existing bag gates.
The additional full-fit audit verifies all four paired trajectories, six
calibration blocks, covariance, costs and variance at `rtol=1e-8`, `atol=1e-9`.
Status, iterations, dimensions and parameter order agree exactly. The final
verdict remains INCONCLUSIVE; supplied GT is unused. This is regression
evidence, with no independent absolute calibration accuracy claim.

```bash
python -m tools.audit_bspline_lidar_imu_benchmark docs/assets/numba_bspline_banded_bag
python docs/assets/numba_bspline_banded_bag/validate-full-fits.py.txt
```

Both audits require CPU dependencies, neither raw sensors, Numba nor CUDA.
The archive includes complete schema-valid results, every per-round fit,
source manifests/archives and exact executed helpers. The auxiliary report
binds its validator and frozen full-fit comparison helper.
The original bag hash is `{protocol['bag_sha256']}`; raw arrays remain external.
See [implementation and limits](../../benchmarks/numba_bspline_planes.md).
"""
(target / 'README.md').write_text(readme, encoding='utf-8', newline='\n')
files = {p.relative_to(target).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest()
         for p in sorted(target.rglob('*')) if p.is_file() and p != target / 'publication-index.json'}
(target / 'publication-index.json').write_text(json.dumps({
    'schema_version': 'slac.bspline_lidar_imu_publication/v0.1',
    'provenance': {'input_protocol_sha256': files['protocol.json']},
    'files': files,
}, indent=2) + '\n', encoding='utf-8', newline='\n')
print(json.dumps({'files': len(files), 'whole_speedup': summary['whole_speedup'],
                  'fit_speedup': summary['fit_speedup']}))
