class Tree(maps._CudaTree):
    def __init__(self,points):
        super().__init__(points)
        self.kernel=cp.RawKernel(query,'query3',options=('--fmad=false',))
    def query(self,points,k):
        distances=cp.empty((len(points),k),dtype=cp.float64)
        indexes=cp.empty((len(points),k),dtype=cp.int64)
        if len(points):
            reorder=len(points)==len(self.data) and points.data.ptr==self.data.data.ptr
            self.kernel(((len(points)+127)//128,),(128,),(
                points,self.data,*self.arrays,np.int32(len(points)),np.int32(len(self.data)),
                np.int32(k),np.int32(reorder),distances,indexes))
            self.query_calls+=1;self.query_rows+=len(points)
        return distances,indexes

def source_order(points,host):
    if mode=='normal':return points
    if mode=='native_order':return cp.asarray(cKDTree(host,balanced_tree=False,compact_nodes=False).indices)
    codes=cp.empty(len(points),dtype=cp.uint64)
    low=host.min(axis=0);high=host.max(axis=0)
    morton_kernel(((len(points)+127)//128,),(128,),(
        points,np.int32(len(points)),*(np.float64(x) for x in (*low,*high)),codes))
    return cp.argsort(codes,kind='stable')
