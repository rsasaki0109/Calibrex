batch_size=4

class Target(adapter.CudaOdometryTarget):
    def register(self,source,initial):
        host=adapter._coordinates(_quantize_geometry(source,self.options.geometry_quantum_m),'source')
        transform=np.array(initial,dtype=np.float64,order='C',copy=True)
        n=len(host);opts=self.options
        if not n or n<len(self.host)/10:return super().register(source,initial)
        self._ensure_all()
        scale=max(self.scale,float(np.max(np.abs(host))))
        with cp.cuda.Device(self.device_id):
            points=cp.asarray(host);matrix=cp.asarray(transform)
            state=cp.zeros(2,dtype=cp.int32)
            meta=cp.asarray(np.array([128*np.finfo(float).eps*scale*max(1.,float(np.linalg.norm(transform,ord=1))),math.inf,0.,0.,0.]))
            hessian=cp.zeros((6,6),dtype=cp.float64)
            terms=cp.empty((31,n),dtype=cp.float64);uncertain=cp.empty(n,dtype=cp.float64)
            moved=cp.empty_like(points);distances=cp.empty((n,2),dtype=cp.float64);indexes=cp.empty((n,2),dtype=cp.int64);pending=cp.empty(n,dtype=cp.int64)
            blocks=(n+255)//256;partial=cp.empty((31,blocks),dtype=cp.float64)
            progress=np.zeros(2,dtype=np.int32)
            while progress[0]==0 and progress[1]<opts.max_iterations:
                count=min(batch_size,opts.max_iterations-int(progress[1]))
                for _ in range(count):
                    query(((n+127)//128,),(128,),(points,matrix,state,self.tree.data,*self.tree.arrays,self._normals,self._planar,self.done,np.int32(n),np.int32(len(self.host)),np.float64(opts.max_correspondence_m),np.float64(opts.huber_delta_m),meta,uncertain,terms,moved,distances,indexes,pending))
                    reduce((31*blocks,),(256,),(terms,np.int32(n),np.int32(blocks),state,partial))
                    solve((1,),(1,),(partial,np.int32(blocks),np.int32(opts.min_correspondences),np.float64(opts.convergence_translation_m),np.float64(opts.convergence_rotation_rad),np.float64(scale),matrix,hessian,meta,state))
                progress=cp.asnumpy(state)
            calls=int(progress[1]);self.tree.query_calls+=calls;self.tree.query_rows+=calls*n;self.fused_calls+=calls
            if progress[0]==2:
                self.batch_fallbacks=getattr(self,'batch_fallbacks',0)+1
                return super().register(source,initial)
            transform=cp.asnumpy(matrix);values=cp.asnumpy(meta);normal=cp.asnumpy(hessian)
        if self._near_motion_boundary(transform):
            self.batch_fallbacks=getattr(self,'batch_fallbacks',0)+1
            return super().register(source,initial)
        eigenvalue=0. if not values[3] else float(np.linalg.eigvalsh(normal)[0])/values[3]
        return ScanRegistration(_canonicalize_motion(transform,opts),bool(values[4] and values[2]>=opts.min_correspondences),calls,int(values[2]),float(values[1]),float(eigenvalue),execution_backend=self.identity)

        for case,(data,opts) in enumerate(cases):
            start=time.perf_counter();target=cls(data['target'],opts);init=time.perf_counter()

            payload={k:v for k,v in vars(result).items() if k!='execution_backend'};payload['transform']=result.transform.tolist()
