
template<int K> __device__ bool broad_query(
    const double* p,const double* reference,const float* reference32,
    const long long* order,const int* axis,const int* child,const long long* span,
    const float* boxes32,const float* split32,int m,double reference_error,
    double* exact,long long* chosen) {
    const int L=K+1;
    float point[3];double cast_error=0;
    for(int a=0;a<3;++a) {
        if(!isfinite(p[a]) || fabs(p[a])>1e6)return false;
        point[a]=__double2float_rn(p[a]);
        cast_error+=fabs((double)point[a]-p[a]);
    }
    float absolute=__double2float_ru((cast_error+reference_error)*
                                    (1.+16.*2.2204460492503131e-16)+1e-18);
    float near[L];long long ids[L];
    for(int j=0;j<L;++j){near[j]=1.f/0.f;ids[j]=m;}
    int stack[64],top=1;stack[0]=0;
    float radius=1.f/0.f;
    while(top) {
        int node=stack[--top];float d[3];
        for(int a=0;a<3;++a)d[a]=fmaxf(0.f,fmaxf(boxes32[6*node+a]-point[a],
                                               point[a]-boxes32[6*node+3+a]));
        float lower=fmaxf(0.f,__fsub_rd(__fmul_rd(sqrtf((d[0]*d[0]+d[1]*d[1])+d[2]*d[2]),
                                                    0.999996f),absolute));
        if(lower>radius)continue;
        if(axis[node]<0) {
            for(long long j=span[2*node];j<span[2*node+1];++j) {
                long long id=order[j];
                float x=point[0]-reference32[3*id],y=point[1]-reference32[3*id+1],
                      z=point[2]-reference32[3*id+2];
                float value=(x*x+y*y)+z*z;
                if(value<near[L-1] || (value==near[L-1] && id<ids[L-1])) {
                    int slot=L-1;
                    while(slot>0 && (value<near[slot-1] ||
                          (value==near[slot-1] && id<ids[slot-1]))) {
                        near[slot]=near[slot-1];ids[slot]=ids[slot-1];--slot;
                    }
                    near[slot]=value;ids[slot]=id;
                    radius=__fadd_ru(__fmul_ru(sqrtf(near[L-1]),1.000004f),absolute);
                }
            }
        } else {
            int side=point[axis[node]]<split32[node]?0:1;
            stack[top++]=child[2*node+1-side];stack[top++]=child[2*node+side];
        }
    }
    for(int j=0;j<K;++j){exact[j]=1./0.;chosen[j]=m;}
    for(int j=0;j<L;++j)if(ids[j]<m) {
        long long id=ids[j];
        double x=p[0]-reference[3*id],y=p[1]-reference[3*id+1],z=p[2]-reference[3*id+2];
        double value=x*x+y*y+z*z;
        if(value<exact[K-1] || (value==exact[K-1] && id<chosen[K-1])) {
            int slot=K-1;
            while(slot>0 && (value<exact[slot-1] ||
                  (value==exact[slot-1] && id<chosen[slot-1]))) {
                exact[slot]=exact[slot-1];chosen[slot]=chosen[slot-1];--slot;
            }
            exact[slot]=value;chosen[slot]=id;
        }
    }
    float cutoff=fmaxf(0.f,__fsub_rd(__fmul_rd(sqrtf(near[L-1]),0.999996f),absolute));
    return isfinite(cutoff) && (double)cutoff>sqrt(exact[K-1]);
}

extern "C" __global__ void project_query_system3(
    const double* xyz, const double* matrix, const double* reference,
    const long long* order, const int* axis, const double* split,
    const int* child, const long long* span, const double* boxes, const double* normal,
    const bool* planar, const bool* done, const int n, const int m, const double gate,
    const double huber, const double error, double* uncertain, double* terms,
    double* moved_out, double* distances_out, long long* indices_out,
    long long* pending_out,
    const float* reference32,const float* boxes32,const float* split32,
    double reference_error,unsigned long long* counters) {
    int row=blockDim.x*blockIdx.x+threadIdx.x;
    if(row>=n) return;
    const double* s=xyz+3*row;
    double p[3];
    for(int a=0;a<3;a++)
        p[a]=matrix[4*a]*s[0]+matrix[4*a+1]*s[1]+matrix[4*a+2]*s[2]+matrix[4*a+3];
    double best[2];long long chosen[2];
    if(!broad_query<2>(p,reference,reference32,order,axis,child,span,boxes32,split32,
                       m,reference_error,best,chosen)) {
        atomicAdd(counters+1,1ULL);
    best[0]=best[1]=1.0/0.0;chosen[0]=chosen[1]=m;
    int stack[64];int top=1;stack[0]=0;
    while(top){
        --top;int node=stack[top];
        double d[3];
        for(int a=0;a<3;a++)
            d[a]=fmax(0.0,fmax(boxes[6*node+a]-p[a],p[a]-boxes[6*node+3+a]));
        double lower=(d[0]*d[0]+d[1]*d[1]+d[2]*d[2])*(1.0-16.0*2.2204460492503131e-16);
        if(lower>best[1]) continue;
        if(axis[node]<0){
            for(long long v=span[2*node];v<span[2*node+1];v++){
                long long position=order[v];
                double x=p[0]-reference[3*position];
                double y=p[1]-reference[3*position+1];
                double z=p[2]-reference[3*position+2];
                double d=x*x+y*y+z*z;
                if(d<best[1] || (d==best[1] && position<chosen[1])){
                    if(d<best[0] || (d==best[0] && position<chosen[0])){
                        best[1]=best[0];chosen[1]=chosen[0];best[0]=d;chosen[0]=position;
                    }else{best[1]=d;chosen[1]=position;}
                }
            }
        }else{
            double delta=p[axis[node]]-split[node];int side=delta<0?0:1;
            if(delta*delta<=best[1]){
                stack[top]=child[2*node+1-side];++top;
            }
            stack[top]=child[2*node+side];++top;
        }
    }
    }
    double distance=sqrt(best[0]);long long hit=chosen[0];
    for(int a=0;a<3;a++) moved_out[3*row+a]=p[a];
    distances_out[2*row]=distance;distances_out[2*row+1]=sqrt(best[1]);
    indices_out[2*row]=hit;indices_out[2*row+1]=chosen[1];
    bool repair=sqrt(best[1])-distance<=error || fabs(distance-gate)<=error;
    uncertain[row]=repair?1.0:0.0;
    bool valid=distance<gate && hit<m;
    bool pending=valid && !done[hit];
    pending_out[row]=pending?hit:-1;
    if(valid) valid=done[hit] && planar[hit];
    double j[6]={0,0,0,0,0,0},r=0,w=1;
    if(valid){
        const double* q=reference+3*hit;const double* a=normal+3*hit;
        r=(p[0]-q[0])*a[0]+(p[1]-q[1])*a[1]+(p[2]-q[2])*a[2];
        j[0]=p[1]*a[2]-p[2]*a[1];j[1]=p[2]*a[0]-p[0]*a[2];
        j[2]=p[0]*a[1]-p[1]*a[0];j[3]=a[0];j[4]=a[1];j[5]=a[2];
        w=fabs(r)<=huber?1.0:huber/fmax(fabs(r),1e-12);
    }
    int column=0;
    for(int a=0;a<6;a++) for(int b=a;b<6;b++)
        terms[(column++)*n+row]=(j[a]*w)*j[b];
    for(int a=0;a<6;a++) terms[(column++)*n+row]=(j[a]*w)*r;
    terms[27*n+row]=r*r;terms[28*n+row]=valid?1.0:0.0;
    terms[29*n+row]=repair?1.0:0.0;
    terms[30*n+row]=pending?1.0:0.0;
}
