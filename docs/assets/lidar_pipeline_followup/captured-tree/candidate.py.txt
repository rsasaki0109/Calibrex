"""Private finite-domain FP32 broad search with FP64 selection and exact fallback."""
import cupy as cp
import numpy as np
from scipy.spatial import cKDTree

from calibrex.solvers import cuda_scan_odometry_adapter as native
from calibrex.solvers.cuda_lidar_map_adapter import _CudaTree
from calibrex.solvers.scipy_spatial_snapshot import capture_spatial_tree


HELPER = r'''
template<int K> __device__ bool broad_query(
    const double* p,const double* reference,const float* reference32,
    const long long* order,const int* axis,const int* child,const long long* span,
    const float* boxes32,const float* split32,int m,double reference_error,
    double* exact,long long* chosen) {
    const int L=K+1;
    float point[3];double cast_error=0;
    for(int a=0;a<3;++a) {
        if(!isfinite(p[a]) || fabs(p[a])>1e6)return false;
        point[a]=__double2float_rn(p[a]);
        cast_error+=fabs((double)point[a]-p[a]);
    }
    float absolute=__double2float_ru((cast_error+reference_error)*
                                    (1.+16.*2.2204460492503131e-16)+1e-18);
    float near[L];long long ids[L];
    for(int j=0;j<L;++j){near[j]=1.f/0.f;ids[j]=m;}
    int stack[64],top=1;stack[0]=0;
    float radius=1.f/0.f;
    while(top) {
        int node=stack[--top];float d[3];
        for(int a=0;a<3;++a)d[a]=fmaxf(0.f,fmaxf(boxes32[6*node+a]-point[a],
                                               point[a]-boxes32[6*node+3+a]));
        float lower=fmaxf(0.f,__fsub_rd(__fmul_rd(sqrtf((d[0]*d[0]+d[1]*d[1])+d[2]*d[2]),
                                                    0.999996f),absolute));
        if(lower>radius)continue;
        if(axis[node]<0) {
            for(long long j=span[2*node];j<span[2*node+1];++j) {
                long long id=order[j];
                float x=point[0]-reference32[3*id],y=point[1]-reference32[3*id+1],
                      z=point[2]-reference32[3*id+2];
                float value=(x*x+y*y)+z*z;
                if(value<near[L-1] || (value==near[L-1] && id<ids[L-1])) {
                    int slot=L-1;
                    while(slot>0 && (value<near[slot-1] ||
                          (value==near[slot-1] && id<ids[slot-1]))) {
                        near[slot]=near[slot-1];ids[slot]=ids[slot-1];--slot;
                    }
                    near[slot]=value;ids[slot]=id;
                    radius=__fadd_ru(__fmul_ru(sqrtf(near[L-1]),1.000004f),absolute);
                }
            }
        } else {
            int side=point[axis[node]]<split32[node]?0:1;
            stack[top++]=child[2*node+1-side];stack[top++]=child[2*node+side];
        }
    }
    for(int j=0;j<K;++j){exact[j]=1./0.;chosen[j]=m;}
    for(int j=0;j<L;++j)if(ids[j]<m) {
        long long id=ids[j];
        double x=p[0]-reference[3*id],y=p[1]-reference[3*id+1],z=p[2]-reference[3*id+2];
        double value=x*x+y*y+z*z;
        if(value<exact[K-1] || (value==exact[K-1] && id<chosen[K-1])) {
            int slot=K-1;
            while(slot>0 && (value<exact[slot-1] ||
                  (value==exact[slot-1] && id<chosen[slot-1]))) {
                exact[slot]=exact[slot-1];chosen[slot]=chosen[slot-1];--slot;
            }
            exact[slot]=value;chosen[slot]=id;
        }
    }
    float cutoff=fmaxf(0.f,__fsub_rd(__fmul_rd(sqrtf(near[L-1]),0.999996f),absolute));
    return isfinite(cutoff) && (double)cutoff>sqrt(exact[K-1]);
}
'''


def sources():
    query = native._SOURCE
    header = query[:query.index('    int row =')]
    header = header.replace('long long* index) {', '''long long* index,
    const float* reference32,const float* boxes32,const float* split32,
    double reference_error,unsigned char* fallback,unsigned long long* counters) {''')
    query = HELPER + header + r'''
    int row=blockDim.x*blockIdx.x+threadIdx.x;if(row>=n)return;
    double p[3]={xyz[3*row],xyz[3*row+1],xyz[3*row+2]};
    double best[11];long long chosen[11];
    bool certified=broad_query<11>(p,reference,reference32,order,axis,child,span,
                                  boxes32,split32,m,reference_error,best,chosen);
    fallback[row]=certified?0:1;
    if(!certified){atomicAdd(counters,1ULL);return;}
    for(int j=0;j<11;++j){distance[11*row+j]=sqrt(best[j]);index[11*row+j]=chosen[j];}
}
'''
    fallback = native._SOURCE.replace('long long* index) {',
                                      'long long* index,const unsigned char* fallback) {')
    fallback = fallback.replace('    if (row >= n) return;',
                                '    if (row >= n || !fallback[row]) return;')
    fused = native._FUSED_SOURCE
    fused = fused.replace('long long* pending_out) {', '''long long* pending_out,
    const float* reference32,const float* boxes32,const float* split32,
    double reference_error,unsigned long long* counters) {''')
    start = fused.index('    double best[2]=')
    stop = fused.index('    double distance=sqrt(best[0]);')
    old = fused[start:stop]
    prefix = '''    double best[2];long long chosen[2];
    if(!broad_query<2>(p,reference,reference32,order,axis,child,span,boxes32,split32,
                       m,reference_error,best,chosen)) {
        atomicAdd(counters+1,1ULL);
'''
    fused = HELPER + fused[:start] + prefix + old + '    }\n' + fused[stop:]
    # The fallback scope must update the arrays consumed by the original system.
    fused = fused.replace('    double best[2]={1.0/0.0,1.0/0.0};long long chosen[2]={m,m};',
                           '    best[0]=best[1]=1.0/0.0;chosen[0]=chosen[1]=m;')
    return query, fallback, fused


QUERY, FALLBACK, FUSED = sources()


class Tree(_CudaTree):
    def __init__(self, points):
        self.cpu = cKDTree(points, copy_data=True)
        self.data = cp.asarray(points)
        topology, bounds, self.capture_method = capture_spatial_tree(self.cpu)
        axes, splits, children, spans = topology
        self.arrays = (cp.asarray(self.cpu.indices, dtype=cp.int64), cp.asarray(axes),
                       cp.asarray(splits), cp.asarray(children), cp.asarray(spans), cp.asarray(bounds))
        self.kernel = cp.RawKernel(native._SOURCE, 'query3', options=('--fmad=false',))
        rounded = self.cpu.data.astype(np.float32)
        self.reference_error = float(np.sum(np.max(np.abs(self.cpu.data-rounded.astype(np.float64)),axis=0)))
        self.reference32 = cp.asarray(rounded)
        self.boxes32 = cp.asarray(bounds, dtype=cp.float32)
        self.split32 = cp.asarray(splits, dtype=cp.float32)
        self.counts = cp.zeros(2, dtype=cp.uint64)
        self.broad = cp.RawKernel(QUERY, 'query3', options=('--fmad=false',))
        self.fallback = cp.RawKernel(FALLBACK, 'query3', options=('--fmad=false',))
        self.bytes = int(self.data.nbytes + sum(a.nbytes for a in self.arrays)
                         + self.reference32.nbytes + self.boxes32.nbytes + self.split32.nbytes)
        self.query_calls = self.query_rows = 0

    def query(self, points, k):
        if k != 11 or len(self.data)<12:
            return super().query(points,k)
        distance = cp.empty((len(points),k),dtype=cp.float64)
        index = cp.empty((len(points),k),dtype=cp.int64)
        repair = cp.empty(len(points),dtype=cp.uint8)
        if len(points):
            arguments=(points,self.data,*self.arrays,np.int32(len(points)),np.int32(len(self.data)),
                       np.int32(k),distance,index)
            self.broad(((len(points)+127)//128,),(128,),(*arguments,self.reference32,
                        self.boxes32,self.split32,np.float64(self.reference_error),repair,self.counts))
            self.fallback(((len(points)+127)//128,),(128,),(*arguments,repair))
            self.query_calls+=1
            self.query_rows+=len(points)
        return distance,index


class Fast(native.CudaOdometryTarget):
    def __init__(self, target, options):
        native._validate_options(options)
        self._host=native._coordinates(native._quantize_geometry(target,options.geometry_quantum_m),
                                       'odometry target')
        if len(self.host)<3 or np.max(np.abs(self.host))>1e6:
            raise ValueError('private finite-domain candidate requires bounded nonempty target')
        tree=Tree(self.host)
        self._initialize_target(tree,options,max(1.,float(np.max(np.abs(self.host)))),native.execution_identity())
        kernel=cp.RawKernel(FUSED,'project_query_system3',options=('--fmad=false',))
        def launch(grid, block, arguments):
            kernel(grid,block,(*arguments,tree.reference32,tree.boxes32,tree.split32,
                               np.float64(tree.reference_error),tree.counts))
        self.fused_kernel=launch

    def statistics(self):
        values=super().statistics()
        counts=cp.asnumpy(self.tree.counts)
        values['broad_normal_fallback_rows']=int(counts[0])
        values['broad_registration_fallback_rows']=int(counts[1])
        return values
