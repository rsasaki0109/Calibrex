class DeviceTree(broad.radix._CudaTree):
    def __init__(self, points):
        _namespace['__init__'](self,points)
        self.reference_error=3*np.finfo(np.float32).eps*self.scale
        self.reference32=self.data.astype(cp.float32)
        self.boxes32=self.arrays[-1].astype(cp.float32)
        self.split32=self.arrays[2].astype(cp.float32)
        self.counts=cp.zeros(2,dtype=cp.uint64)
        self.broad=cp.RawKernel(QUERY,'query3',options=('--fmad=false',))
        self.fallback=cp.RawKernel(FALLBACK,'query3',options=('--fmad=false',))
        self.bytes+=self.reference32.nbytes+self.boxes32.nbytes+self.split32.nbytes

    @property
    def host(self):
        if self._host is None:self._host=broad.native._coordinates(cp.asnumpy(self.data),'target')
        return self._host

    @property
    def cpu(self):
        _ = self.host
        return super().cpu

    def reserve(self,count):
        pass

    query=broad.Tree.query

class Fast(resident._ResidentTarget):
    def __init__(self, points, options, identity):
        broad.native._validate_options(options)
        tree=DeviceTree(points)
        self.grid=tree
        self._normal_payload=None
        self.sparse_native_normal_rows=self.spatial_native_normal_rows=0
        self._host=None
        self._initialize_target(tree,options,tree.scale,identity)
        kernel=cp.RawKernel(FUSED,'project_query_system3',options=('--fmad=false',))
        def launch(grid,block,arguments):
            kernel(grid,block,(*arguments,tree.reference32,tree.boxes32,tree.split32,
                               np.float64(tree.reference_error),tree.counts))
        self.fused_kernel=launch

    def statistics(self):
        values=broad.native.CudaOdometryTarget.statistics(self)
        counts=cp.asnumpy(self.tree.counts)
        values.update({'broad_normal_fallback_rows':int(counts[0]),
                       'broad_registration_fallback_rows':int(counts[1]),
                       'device_tree_build_kernel_calls':self.tree.build_kernel_calls,
                       'native_tree_builds':self.tree.native_tree_builds,
                       'sparse_native_normal_rows':self.sparse_native_normal_rows,
                       'spatial_native_normal_rows':self.spatial_native_normal_rows})
        return values
