import hashlib
import json
import statistics
import subprocess
from pathlib import Path

from calibrex.core.benchmark import BenchmarkDefinition, aggregate_benchmark_definition
from calibrex.core.validation import validate_file
from tools.benchmark_livox_evaluation import SOURCES

ROOT = Path('outputs/public-livox-evaluation-v2')
DEST = Path('docs/assets/livox_evaluation')
COMMIT = subprocess.check_output(['git', 'rev-parse', '8c760c7'], text=True).strip()
BASELINE = 'c6fd98b7b5af73be41fa01b5e4938773f4696fb1'
WORKSPACE = str(Path.cwd())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def normalize(value):
    if isinstance(value, dict):
        return {normalize(k): normalize(v) for k, v in value.items()}
    if isinstance(value, list):
        return [normalize(v) for v in value]
    if isinstance(value, str):
        value = value.replace(WORKSPACE, '${WORKSPACE_ROOT}').replace(
            WORKSPACE.replace('//', '/'), '${WORKSPACE_ROOT}')
        if '${WORKSPACE_ROOT}' in value:
            value = value.replace('//', '/')
    return value


DEST.mkdir(parents=True, exist_ok=True)
bindings = {}
for name in ('livox', 'lidar'):
    path = ROOT / f'baseline.{name}.py.txt'
    target = DEST / path.name
    target.write_bytes(path.read_bytes())
    bindings['published:' + target.name] = sha(target)
for source in SOURCES:
    target = DEST / ('measured.' + Path(source).name + '.txt')
    data = subprocess.check_output(['git', 'show', COMMIT + ':' + source])
    target.write_bytes(data)
    bindings['published:' + target.name] = sha(target)
    bindings['committed:' + source] = sha(target)
for filename, source in (
    ('publication-launcher.py.txt', Path(__file__)),
    ('profiling-launcher.py.txt', Path('.tmp-calibrex-profile-livox-pipeline.py')),
    ('pilot-import-failure.log.txt', Path('outputs/public-livox-evaluation-pilot-v1/baseline.repeat-01.log')),
):
    target = DEST / filename
    target.write_text(normalize(source.read_text(encoding='utf-8')).rstrip() + '/n', encoding='utf-8', newline='/n')
    bindings['published:' + target.name] = sha(target)

total = 0
definitions = {}
for prefix, root, expected in (
    ('', ROOT, 10),
    ('pilot.', Path('outputs/public-livox-evaluation-pilot-v2'), 2),
):
    raw = BenchmarkDefinition.model_validate_json((root / 'pipeline.definition.json').read_bytes())
    assert len(raw.trials) == expected
    assert all(t.status == 'success' for t in raw.trials)
    anchor = raw.trials[0].provenance.notes[-1]
    for trial in raw.trials:
        encoded = trial.provenance.notes[-1]
        assert hashlib.sha256(encoded.encode()).hexdigest() == trial.provenance.output_sha256
        assert encoded == anchor
        metadata = json.loads(trial.provenance.notes[0])
        directory = root / f'{trial.method_id}.repeat-{metadata["repeat"]:02d}'
        assert sha(directory / 'result.yaml') == metadata['result_sha256']
        assert validate_file(directory / 'result.yaml').valid
        for filename, digest in metadata['artifacts'].items():
            assert sha(directory / filename) == digest
        total += 1
    published = normalize(raw.model_dump(mode='json'))
    for trial, original in zip(published['trials'], raw.trials, strict=True):
        trial['provenance']['notes'][0] = json.dumps(
            normalize(json.loads(original.provenance.notes[0])), sort_keys=True)
        # Preserve the entire canonical result string and its original digest.
        trial['provenance']['notes'][-1] = original.provenance.notes[-1]
    published['provenance']['source_artifacts'].update(bindings)
    published['provenance']['source_artifacts']['original:definition'] = sha(root / 'pipeline.definition.json')
    published['provenance']['source_artifacts']['original:aggregate'] = sha(root / 'pipeline.benchmark.json')
    if prefix:
        published['methods'][1]['source_commit'] = None
        published['limitations'].append(
            'Pilot candidate was uncommitted and preceded batched coordinate transforms; '
            'working-byte digests are retained but no final-commit timing is attributed to it.')
    published['limitations'].append(
        'Publication normalizes workspace paths in commands and metadata; complete normalized '
        'result strings, timings and all original input/config/output/report digests are retained.')
    published['limitations'].append(
        'Fresh workers call the public calibration Python API; calibrex CLI argument dispatch '
        'is excluded. Worker measurement JSON writes are included in subprocess wall.')
    definition = BenchmarkDefinition.model_validate(published)
    definition.save(DEST / f'{prefix}pipeline.definition.json')
    aggregate_benchmark_definition(definition).save(DEST / f'{prefix}pipeline.benchmark.json')
    definitions[prefix] = definition
for path in DEST.glob('*.json'):
    assert validate_file(path).valid
assert total == 12

definition = definitions['']
rows = []
for repeat in range(1, 6):
    pair = {t.method_id: t for t in definition.trials if t.split_id == f'repeat-{repeat:02d}'}
    old, new = pair['baseline'], pair['batched']
    rows.append(f'| {repeat} | {old.runtime_seconds:.6f} | {new.runtime_seconds:.6f} | '
                f'{old.metrics["pipeline_seconds"]:.6f} | {new.metrics["pipeline_seconds"]:.6f} |')
medians = {method: {name: statistics.median(
    t.metrics[name] for t in definition.trials if t.method_id == method)
    for name in ('subprocess_wall_seconds', 'pipeline_seconds')}
    for method in ('baseline', 'batched')}
ranges = {method: [(min(t.metrics[name] for t in definition.trials if t.method_id == method),
                    max(t.metrics[name] for t in definition.trials if t.method_id == method))
                   for name in ('subprocess_wall_seconds', 'pipeline_seconds')]
          for method in ('baseline', 'batched')}
wall_ratio = medians['baseline']['subprocess_wall_seconds'] / medians['batched']['subprocess_wall_seconds']
pipeline_ratio = medians['baseline']['pipeline_seconds'] / medians['batched']['pipeline_seconds']

## Numerical contract and retained evidence

The same lexicographic 27-neighbor order resolves ties. A nearest voxel without
a normal still rejects a correspondence. Residual multiply/add/sqrt ordering,
Hamilton product ordering, input-order sums, population denominators and scalar
exception behavior are preserved. Query chunks materialize at most 4,096 × 27
associations. Very large keys and unsupported coordinates use the original
scalar path. Every evaluation invocation constructs a new context and observes
fresh inputs; reused geometry is confined to that invocation.

`pipeline.definition.json` and `pipeline.benchmark.json` retain all ten timings,
all complete result strings and their canonical output SHA256 values. Raw result
file and every written artifact's digests are separate and remain recorded.
Parity normalizes only run ID, creation/generation timestamps, destination paths
and Windows path separators. All numeric and diagnostic fields remain present.
The four JSON reports validate against standard benchmark schemas, and both
aggregates are recomputed exactly from their published definitions.

`pilot.pipeline.*` retains the two earlier full-pipeline trials: 61.149350 s
baseline and 26.448565 s candidate, with exact within-pilot result parity. That
uncommitted candidate preceded batched coordinate transforms and is not bound to
the final implementation commit. No pilot timing enters the final medians.
`pilot-import-failure.log.txt` retains an earlier launcher import-order failure;
it produced no calibration trial. The corrected loader uses normal package
initialization order. Source/archive digests and the publication/profiling
launchers remain available. Published measured/baseline sources are exact Git blobs; original
working-byte hashes and original report hashes are recorded separately.

A baseline cProfile investigation measured 174.301498 s with profiling overhead;
166.427 s lay in candidate/control evaluation. Its nested cumulative costs
included 47.052 s neighborhood residuals, 37.463 s PCD decoding, 27.730 s repeated
geometry coverage and 27.679 s repeated coordinate transforms. These are
overlapping profiling costs, not formal wall timings or additional speedups.
The profiling launcher and baseline sources bind that investigation; its
instrumented timing does not enter any benchmark median.
