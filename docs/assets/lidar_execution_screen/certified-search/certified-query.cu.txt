
__device__ float lower3(const float* a, const float* b) {
    float d[3];
    for(int axis=0;axis<3;++axis)
        d[axis]=fmaxf(0.f,fmaxf(__fsub_rd(a[axis],b[3+axis]),
                             __fsub_rd(b[axis],a[3+axis])));
    float v=__fadd_rd(__fadd_rd(__fmul_rd(d[0],d[0]),__fmul_rd(d[1],d[1])),
                            __fmul_rd(d[2],d[2]));
    return __fmul_rd(v,0.999999f);
}
__device__ float upper3(const float* a, const float* b) {
    float d[3];
    for(int axis=0;axis<3;++axis)
        d[axis]=fmaxf(__fsub_ru(a[3+axis],b[axis]),__fsub_ru(b[3+axis],a[axis]));
    float v=__fadd_ru(__fadd_ru(__fmul_ru(d[0],d[0]),__fmul_ru(d[1],d[1])),
                            __fmul_ru(d[2],d[2]));
    return __fmul_ru(v,1.000001f);
}

extern "C" __global__ void query3(
    const double* xyz, const double* reference, const long long* order,
    const int* axis, const double* split, const int* child,
    const long long* span, const double* boxes, const int n, const int m, const int k,
    double* distance, long long* index, const float* point_bounds,
    const float* box_bounds, const float* float_splits) {
    int row=blockDim.x*blockIdx.x+threadIdx.x;
    if(row>=n)return;
    double p[3]={xyz[3*row],xyz[3*row+1],xyz[3*row+2]};
    float interval[6],center[3];
    for(int a=0;a<3;++a) {
        interval[a]=__double2float_rd(p[a]);
        interval[3+a]=__double2float_ru(p[a]);
        center[a]=__double2float_rn(p[a]);
    }
    float hi[17],lo[17],discarded=1.f/0.f;
    double best[17]; long long chosen[17];
    for(int j=0;j<k;++j){hi[j]=1.f/0.f;lo[j]=1.f/0.f;chosen[j]=m;}
    int stack[64],top=1;stack[0]=0;
    while(top) {
        int node=stack[--top];
        if(lower3(interval,box_bounds+6*node)>hi[k-1])continue;
        if(axis[node]<0) {
            for(long long j=span[2*node];j<span[2*node+1];++j) {
                long long position=order[j];
                float low=lower3(interval,point_bounds+6*position);
                if(low>hi[k-1])continue;
                float high=upper3(interval,point_bounds+6*position);
                if(high<hi[k-1] || (high==hi[k-1] && position<chosen[k-1])) {
                    if(chosen[k-1]<m)discarded=fminf(discarded,lo[k-1]);
                    int slot=k-1;
                    while(slot>0 && (high<hi[slot-1] ||
                           (high==hi[slot-1] && position<chosen[slot-1]))) {
                        hi[slot]=hi[slot-1];lo[slot]=lo[slot-1];chosen[slot]=chosen[slot-1];--slot;
                    }
                    hi[slot]=high;lo[slot]=low;chosen[slot]=position;
                } else discarded=fminf(discarded,low);
            }
        } else {
            int side=center[axis[node]]<float_splits[node]?0:1;
            stack[top++]=child[2*node+1-side];stack[top++]=child[2*node+side];
        }
    }
    // Refine every selected candidate with the original FP64 distance arithmetic.
    for(int j=0;j<k;++j) {
        double value=1./0.;long long position=chosen[j];
        if(position<m) {
            double x=p[0]-reference[3*position],y=p[1]-reference[3*position+1],
                   z=p[2]-reference[3*position+2];
            value=x*x+y*y+z*z;
        }
        int slot=j;
        while(slot>0 && (value<best[slot-1] ||
                    (value==best[slot-1] && position<chosen[slot-1]))) {
            best[slot]=best[slot-1];chosen[slot]=chosen[slot-1];--slot;
        }
        best[slot]=value;chosen[slot]=position;
    }
    // An unselected interval that overlaps the exact cutoff triggers the entire
    // original search in this thread; no approximate membership is accepted.
    if(!isfinite(discarded) || (double)discarded<=best[k-1]) {
    for(int j=0;j<k;j++){best[j]=1.0/0.0;chosen[j]=m;}
    top=1;
    stack[0]=0;
    while(top){
        --top; int node=stack[top];
        double d[3];
        for(int a=0;a<3;a++)
            d[a]=fmax(0.0,fmax(boxes[6*node+a]-p[a],p[a]-boxes[6*node+3+a]));
        // Retain equal and roundoff-close boxes; candidate distance arithmetic is unchanged.
        double lower=(d[0]*d[0]+d[1]*d[1]+d[2]*d[2])*(1.0-16.0*2.2204460492503131e-16);
        if(lower > best[k-1]) continue;
        if(axis[node]<0){
            for(long long j=span[2*node];j<span[2*node+1];j++){
                long long position=order[j];
                double x=p[0]-reference[3*position];
                double y=p[1]-reference[3*position+1];
                double z=p[2]-reference[3*position+2];
                double d=x*x+y*y+z*z;
                if(d<best[k-1] || (d==best[k-1] && position<chosen[k-1])){
                    int slot=k-1;
                    while(slot>0 && (d<best[slot-1] ||
                         (d==best[slot-1] && position<chosen[slot-1]))){
                        best[slot]=best[slot-1];chosen[slot]=chosen[slot-1];--slot;
                    }
                    best[slot]=d;chosen[slot]=position;
                }
            }
        }else{
            double delta=p[axis[node]]-split[node];
            int side=delta<0?0:1;
            int far=child[2*node+1-side], near=child[2*node+side];
            if(delta*delta <= best[k-1]){
                stack[top]=far;++top;
            }
            stack[top]=near;++top;
        }
    }
    }
    for(int j=0;j<k;j++){distance[row*k+j]=sqrt(best[j]);index[row*k+j]=chosen[j];}
}
