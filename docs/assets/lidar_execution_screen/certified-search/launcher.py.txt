"""Spent public GLIM: certified FP32 candidate intervals with exact FP64 selection/fallback."""
import os
from pathlib import Path
ROOT = Path(__file__).resolve().parent
OUT = ROOT / "outputs/certified-fp32-search-pilot-v1"
for key in ("OPENBLAS_NUM_THREADS", "MKL_NUM_THREADS", "OMP_NUM_THREADS"):
    os.environ[key] = "1"
os.environ["CUPY_CACHE_DIR"] = str(OUT / "cupy-cache")
os.environ["CUDA_CACHE_DISABLE"] = "1"
os.environ["PYTHONDONTWRITEBYTECODE"] = "1"

from dataclasses import asdict
import hashlib
import json
import re
import shutil
import statistics
import sys
import time

SOURCE = ROOT / "outputs/cpu-map-bag-comparison-v1/source-current"
FIXTURES = ROOT / "outputs/odometry-search-profile-v3"
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()


def write(path, value):
    path.write_text(json.dumps(value, indent=2, allow_nan=False) + "\n", encoding="utf-8")


def main():
    OUT.mkdir(exist_ok=False)
    shutil.copyfile(Path(__file__), OUT / "launcher.py.txt")
    from tools.benchmark_bspline_lidar_imu import _verify_sources
    expected = _verify_sources(SOURCE)
    sys.path.insert(0, str(SOURCE / "src"))
    import cupy as cp
    import numpy as np
    from calibrex.solvers import cuda_scan_odometry_adapter as cuda
    from calibrex.solvers.scan_to_scan_odometry import ScanOdometryOptions

    from calibrex.solvers.cuda_lidar_map_adapter import _CudaTree

    helpers = r"""
__device__ float lower3(const float* a, const float* b) {
    float d[3];
    for(int axis=0;axis<3;++axis)
        d[axis]=fmaxf(0.f,fmaxf(__fsub_rd(a[axis],b[3+axis]),
                             __fsub_rd(b[axis],a[3+axis])));
    float v=__fadd_rd(__fadd_rd(__fmul_rd(d[0],d[0]),__fmul_rd(d[1],d[1])),
                            __fmul_rd(d[2],d[2]));
    return __fmul_rd(v,0.999999f);
}
__device__ float upper3(const float* a, const float* b) {
    float d[3];
    for(int axis=0;axis<3;++axis)
        d[axis]=fmaxf(__fsub_ru(a[3+axis],b[axis]),__fsub_ru(b[3+axis],a[axis]));
    float v=__fadd_ru(__fadd_ru(__fmul_ru(d[0],d[0]),__fmul_ru(d[1],d[1])),
                            __fmul_ru(d[2],d[2]));
    return __fmul_ru(v,1.000001f);
}
"""
    header = r"""
extern "C" __global__ void query3(
    const double* xyz, const double* reference, const long long* order,
    const int* axis, const double* split, const int* child,
    const long long* span, const double* boxes, const int n, const int m, const int k,
    double* distance, long long* index, const float* point_bounds,
    const float* box_bounds, const float* float_splits) {
    int row=blockDim.x*blockIdx.x+threadIdx.x;
    if(row>=n)return;
    double p[3]={xyz[3*row],xyz[3*row+1],xyz[3*row+2]};
    float interval[6],center[3];
    for(int a=0;a<3;++a) {
        interval[a]=__double2float_rd(p[a]);
        interval[3+a]=__double2float_ru(p[a]);
        center[a]=__double2float_rn(p[a]);
    }
    float hi[17],lo[17],discarded=1.f/0.f;
    double best[17]; long long chosen[17];
    for(int j=0;j<k;++j){hi[j]=1.f/0.f;lo[j]=1.f/0.f;chosen[j]=m;}
    int stack[64],top=1;stack[0]=0;
    while(top) {
        int node=stack[--top];
        if(lower3(interval,box_bounds+6*node)>hi[k-1])continue;
        if(axis[node]<0) {
            for(long long j=span[2*node];j<span[2*node+1];++j) {
                long long position=order[j];
                float low=lower3(interval,point_bounds+6*position);
                if(low>hi[k-1])continue;
                float high=upper3(interval,point_bounds+6*position);
                if(high<hi[k-1] || (high==hi[k-1] && position<chosen[k-1])) {
                    if(chosen[k-1]<m)discarded=fminf(discarded,lo[k-1]);
                    int slot=k-1;
                    while(slot>0 && (high<hi[slot-1] ||
                           (high==hi[slot-1] && position<chosen[slot-1]))) {
                        hi[slot]=hi[slot-1];lo[slot]=lo[slot-1];chosen[slot]=chosen[slot-1];--slot;
                    }
                    hi[slot]=high;lo[slot]=low;chosen[slot]=position;
                } else discarded=fminf(discarded,low);
            }
        } else {
            int side=center[axis[node]]<float_splits[node]?0:1;
            stack[top++]=child[2*node+1-side];stack[top++]=child[2*node+side];
        }
    }
    // Refine every selected candidate with the original FP64 distance arithmetic.
    for(int j=0;j<k;++j) {
        double value=1./0.;long long position=chosen[j];
        if(position<m) {
            double x=p[0]-reference[3*position],y=p[1]-reference[3*position+1],
                   z=p[2]-reference[3*position+2];
            value=x*x+y*y+z*z;
        }
        int slot=j;
        while(slot>0 && (value<best[slot-1] ||
                    (value==best[slot-1] && position<chosen[slot-1]))) {
            best[slot]=best[slot-1];chosen[slot]=chosen[slot-1];--slot;
        }
        best[slot]=value;chosen[slot]=position;
    }
    // An unselected interval that overlaps the exact cutoff triggers the entire
    // original search in this thread; no approximate membership is accepted.
    if(!isfinite(discarded) || (double)discarded<=best[k-1]) {
"""
    native = cuda._SOURCE
    native = native[native.index('    for(int j=0;j<k;j++){best'):native.index('    for(int j=0;j<k;j++){distance')]
    native = native.replace('    int stack[64]; int top=1;', '    top=1;')
    output = '    }\n    for(int j=0;j<k;j++){distance[row*k+j]=sqrt(best[j]);index[row*k+j]=chosen[j];}\n}\n'
    fused = helpers + header + native + output
    (OUT / 'certified-query.cu.txt').write_text(fused, encoding='utf-8')

    def intervals(values):
        rounded = np.asarray(values, dtype=np.float32)
        return np.concatenate((np.nextafter(rounded, np.float32(-np.inf)),
                               np.nextafter(rounded, np.float32(np.inf))), axis=-1)

    class Tree(_CudaTree):
        def __init__(self, points):
            super().__init__(points)
            self.point_intervals = cp.asarray(intervals(self.cpu.data))
            native_bounds = cp.asnumpy(self.arrays[-1]).reshape(-1, 2, 3)
            self.box_intervals = cp.asarray(np.concatenate((
                np.nextafter(native_bounds[:, 0].astype(np.float32), np.float32(-np.inf)),
                np.nextafter(native_bounds[:, 1].astype(np.float32), np.float32(np.inf))), axis=1))
            self.float_splits = self.arrays[2].astype(cp.float32)
            self.kernel = cp.RawKernel(fused, 'query3', options=('--fmad=false',))

        def query(self, points, k):
            assert 1 <= k <= 17
            distances = cp.empty((len(points), k), dtype=cp.float64)
            indexes = cp.empty((len(points), k), dtype=cp.int64)
            if len(points):
                self.kernel(((len(points)+127)//128,), (128,), (
                    points, self.data, *self.arrays, np.int32(len(points)),
                    np.int32(len(self.data)), np.int32(k), distances, indexes,
                    self.point_intervals, self.box_intervals, self.float_splits))
                self.query_calls += 1
                self.query_rows += len(points)
            return distances, indexes

    class Fused(cuda.CudaOdometryTarget):
        def __init__(self, target, options):
            cuda._validate_options(options)
            self._host = cuda._coordinates(cuda._quantize_geometry(target, options.geometry_quantum_m),
                                            'odometry target')
            self._initialize_target(Tree(self.host), options,
                                    max(1., float(np.max(np.abs(self.host)))), cuda.execution_identity())

    variants = {"prior": cuda.CudaOdometryTarget, "fused": Fused}
    cases = []
    for number in range(8):
        path = FIXTURES / f"registration-{number:02d}.npz"
        metadata = json.loads(path.with_suffix(".json").read_bytes())
        assert sha(path) == metadata["fixture_sha256"]
        cases.append((np.load(path), ScanOdometryOptions(**metadata["options"])))
    protocol = {
        "scope": __doc__, "launcher_sha256": sha(Path(__file__)),
        "source_commit": "917748b10d5eff9bb077b332a018eafdb33e9fcd",
        "source_manifest_sha256": sha(SOURCE / "manifest.json"),
        "candidate_kernel_sha256": sha(OUT / "certified-query.cu.txt"),
        "fixtures_sha256": {p.name: sha(p) for p in sorted(FIXTURES.glob("registration-*.*"))},
        "blas_threads": 1, "cuda_cache_initially_empty": not (OUT / "cupy-cache").exists(),
        "limitations": ["Six alternating warm component repeats after one retained warmup case per arm.",
                        "Target construction and registration; not complete bag/calibration.",
                        "CPU/GPU unreserved; no accuracy or SOTA claim.",
                        "Experimental kernel separately source-bound; ordinary runtime identity is not a fused-kernel identity."],
    }
    write(OUT / "protocol.json", protocol)
    runs, references, warmups = [], {}, []

    def execute(variant, case, repeat):
        fixture, options = cases[case]
        started = time.perf_counter()
        target = variants[variant](fixture["target"], options)
        captured = time.perf_counter()
        result = target.register(fixture["source"], fixture["initial"])
        elapsed = time.perf_counter() - started
        values = asdict(result)
        values["transform"] = values["transform"].tolist()
        if case in references:
            previous = references[case]
            assert np.array_equal(np.asarray(previous["transform"]).view(np.uint64),
                                  np.asarray(values["transform"]).view(np.uint64)), (variant, case, "pose")
            assert {k: v for k, v in previous.items() if k != "transform"} == {
                k: v for k, v in values.items() if k != "transform"}, (variant, case, "quality")
        else:
            references[case] = values
        return {"variant": variant, "case": case, "repeat": repeat, "seconds": elapsed,
                "target_seconds": captured - started, "registration_seconds": elapsed - (captured - started),
                "result": values, "statistics": target.statistics()}

    for variant in variants:
        warmups.append(execute(variant, 0, -1))
        write(OUT / "warmups.json", warmups)
    names = tuple(variants)
    for repeat in range(6):
        for variant in names[repeat % 2:] + names[:repeat % 2]:
            for case in range(8):
                runs.append(execute(variant, case, repeat))
                write(OUT / "runs.json", runs)
            print(json.dumps({"repeat": repeat, "variant": variant,
                              "seconds": sum(r["seconds"] for r in runs
                                             if r["repeat"] == repeat and r["variant"] == variant)}), flush=True)
    _verify_sources(SOURCE)
    for name, module in list(sys.modules.items()):
        if name == "calibrex" or name.startswith("calibrex."):
            path = Path(module.__file__).resolve()
            assert path.is_relative_to(SOURCE / "src")
            assert sha(path) == expected[path.relative_to(SOURCE / "src").as_posix()]
    medians = {variant: statistics.median(sum(r["seconds"] for r in runs
                                            if r["variant"] == variant and r["repeat"] == repeat)
                                         for repeat in range(6)) for variant in names}
    summary = {"observations": len(runs), "all_registration_float_bits_equal": True,
               "medians_seconds": medians, "ratio": medians["prior"] / medians["fused"],
               "protocol_sha256": sha(OUT / "protocol.json"), "runs_sha256": sha(OUT / "runs.json")}
    write(OUT / "summary.json", summary)
    print(json.dumps(summary), flush=True)


if __name__ == "__main__":
    main()
