import os
from pathlib import Path
import hashlib
import json
import time

root = Path("outputs/pair-kernel-probe-v1")
root.mkdir()
os.environ["CUPY_CACHE_DIR"] = str((root/"cache").resolve())
for key in ("OPENBLAS_NUM_THREADS", "OMP_NUM_THREADS", "MKL_NUM_THREADS"):
    os.environ[key] = "1"
import cupy as cp
import numpy as np
from calibrex.solvers.cuda_lidar_map_adapter import _CudaTree
from calibrex.solvers.cuda_point_to_point_icp_adapter import _CudaPairTree, _PAIR_SOURCE
from tools.benchmark_cuda_point_to_point_icp import _identity

dataset = Path("data/public/kitti_raw_icp/2011_09_30_drive_0018_sync/velodyne_points/data")
raw = [(dataset/name).read_bytes() for name in ("0000000115.bin","0000000116.bin")]
query, reference = [np.frombuffer(r,dtype=np.float32).reshape(-1,4)[:,:3].astype(np.float64) for r in raw]
device = cp.asarray(query)
trees = {"generic":_CudaTree(reference),"specialized":_CudaPairTree(reference)}
trials = []
anchor = None
for repeat in range(1,13):
    order = ["generic","specialized"] if repeat%2 else ["specialized","generic"]
    for method in order:
        start = time.perf_counter()
        before, after = cp.cuda.Event(), cp.cuda.Event()
        before.record()
        distances, indices = trees[method].query(device,2)
        after.record()
        after.synchronize()
        seconds = time.perf_counter()-start
        gpu_ms = cp.cuda.get_elapsed_time(before,after)
        actual = (cp.asnumpy(indices),cp.asnumpy(distances).view(np.uint64))
        if anchor is None:
            anchor = actual
        else:
            assert all(np.array_equal(x,y) for x,y in zip(anchor,actual))
        trials.append({"repeat":repeat,"method":method,"wall_seconds":seconds,"event_ms":gpu_ms})
record = {"provenance":{"generator":"cuda_pair_kernel_probe/v1","source":_identity(),
    "launcher_sha256":hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    "input_sha256":[hashlib.sha256(r).hexdigest() for r in raw],
    "pair_source_sha256":hashlib.sha256(_PAIR_SOURCE.encode()).hexdigest(),
    "scope":"kernel-only actual public frame queries; first cold compilations retained"},
    "trials":trials,"proposal_and_distance_bits_equal":True,
    "stack_capacity":trees["specialized"].stack_capacity}
(root/"probe.json").write_text(json.dumps(record,indent=2)+"\n")
print(json.dumps({"medians_seconds":{method:float(np.median([t["wall_seconds"] for t in trials if t["method"]==method])) for method in trees},"exact_equal":True,"stack_capacity":record["stack_capacity"]}))
