"""Reproduce an unrelated screw-linear assertion against frozen prior sources."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

from tools.benchmark_bspline_lidar_imu import _freeze

root = Path(__file__).resolve().parent
out = root / 'outputs/holdout-reuse-screw-baseline-v1'
out.mkdir(exist_ok=False)
source = out / 'sources-prior'
commit = 'e59600c1354dc14f0b5afd06f92d90bf61d56f41'
_freeze(source, commit)
environment = os.environ.copy()
environment['PYTHONPATH'] = str(source / 'src')
for name in ('OPENBLAS_NUM_THREADS', 'OMP_NUM_THREADS', 'MKL_NUM_THREADS'):
    environment[name] = '1'
code = '''import pathlib, pytest
import calibrex.evaluation.continuous_time_lidar_imu_joint as module
assert pathlib.Path(module.__file__).is_relative_to(pathlib.Path("outputs/holdout-reuse-screw-baseline-v1/sources-prior/src").resolve())
print("Frozen module:", module.__file__, flush=True)
raise SystemExit(pytest.main(["tests/unit/test_continuous_time_lidar_imu_joint.py::test_joint_recovery_passes_with_provenance_and_valid_schema[screw_linear]", "--basetemp", "outputs/holdout-reuse-screw-baseline-v1/tmp", "-p", "no:cacheprovider", "--override-ini", "addopts=--disable-plugin-autoload", "-q"]))
'''
command = [sys.executable, '-c', code]
with (out / 'validation.log').open('w', encoding='utf-8') as log:
    result = subprocess.run(command, env=environment, stdout=log, stderr=subprocess.STDOUT, text=True)
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
receipt = {'scope': __doc__, 'source_commit': commit, 'command': command, 'exit_code': result.returncode,
           'source_manifest_sha256': sha(source / 'manifest.json'), 'log_sha256': sha(out / 'validation.log'),
           'launcher_sha256': sha(Path(__file__)), 'blas_threads': 1}
(out / 'receipt.json').write_text(json.dumps(receipt, indent=2) + '\n', encoding='utf-8')
print((out / 'validation.log').read_text(encoding='utf-8'), flush=True)
print(json.dumps(receipt), flush=True)
