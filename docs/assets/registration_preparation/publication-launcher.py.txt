"""Publish all registration timings and preserve the exploratory attempts."""
import hashlib
import json
import os
import subprocess
from pathlib import Path

from calibrex.core.benchmark import BenchmarkDefinition, aggregate_benchmark_definition
from calibrex.core.provenance import sha256_path
from calibrex.core.validation import validate_file

COMMIT = '41c820f4f55e8ead78dd367c33c7194f42758ad9'
BASELINE = '9fe0d4f6ff866bed771b62ccbe7b7e57924b12bb'
ROOT = Path('outputs/public-kitti-registration-preparation-v1')
DEST = Path('docs/assets/registration_preparation')
PREFIXES = ((str(Path.cwd()), '${WORKSPACE_ROOT}'),
            (str(Path(os.environ['CALIBREX_DATASET_ROOT'])), '${DATASET_ROOT}'))
SOURCES = (
    'src/calibrex/solvers/robust_point_to_point_icp_solver.py',
    'src/calibrex/solvers/chen_medioni_point_to_plane_icp_solver.py',
    'src/calibrex/solvers/native_registration_comparison_solver.py',
    'src/calibrex/solvers/_exact_neighbors.py',
    'src/calibrex/core/geometry.py',
    'src/calibrex/evaluation/holdout.py',
    'tools/benchmark_registration_preparation.py',
)


def normalize(value):
    if isinstance(value, dict):
        return {normalize(k): normalize(v) for k, v in value.items()}
    if isinstance(value, list):
        return [normalize(v) for v in value]
    if not isinstance(value, str):
        return value
    for original, replacement in PREFIXES:
        value = value.replace(original, replacement).replace(original.replace('\\', '/'), replacement)
    if any(replacement in value for _, replacement in PREFIXES):
        value = value.replace('\\', '/')
    return value


def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(',', ':'),
                                    allow_nan=False).encode()).hexdigest()


DEST.mkdir(parents=True, exist_ok=True)
bindings = {}
for path in sorted(ROOT.glob('baseline.*.txt')):
    target = DEST / path.name
    target.write_bytes(path.read_bytes())
    bindings[target.name] = sha256_path(target)
for source in SOURCES:
    target = DEST / ('measured.' + Path(source).name + '.txt')
    target.write_bytes(subprocess.check_output(['git', 'show', COMMIT + ':' + source]))
    bindings[target.name] = sha256_path(target)
launcher = DEST / 'publication-launcher.py.txt'
launcher.write_bytes(Path(__file__).read_bytes())
bindings[launcher.name] = sha256_path(launcher)

final = BenchmarkDefinition.model_validate_json((ROOT / 'registration.definition.json').read_bytes())
anchors = {}
full_payloads = 0
for prefix, root, expected_count in (
    ('', ROOT, 90),
    ('pilot-v1.', Path('outputs/public-kitti-registration-preparation-tool-pilot-v1'), 6),
    ('pilot-v2.', Path('outputs/public-kitti-registration-preparation-tool-pilot-v2'), 6),
):
    original = BenchmarkDefinition.model_validate_json((root / 'registration.definition.json').read_bytes())
    definition = BenchmarkDefinition.model_validate(normalize(original.model_dump(mode='json')))
    assert len(definition.trials) == expected_count
    for trial in definition.trials:
        payload = json.loads(trial.provenance.notes[-1].removeprefix('Complete numerical payload: '))
        assert digest(payload) == trial.provenance.output_sha256
        case = trial.split_id.rsplit('-repeat-', 1)[0]
        if prefix == 'pilot-v1.' and trial.method_id == 'vectorized_sequential':
            assert trial.status == 'failed' and trial.metrics['numerical_parity'] == 0.
            assert payload['point_to_plane']['resolved_options']['common_icp']['multi_start_workers'] == 1
            comparison = json.loads(json.dumps(payload))
            comparison['point_to_plane']['resolved_options']['common_icp']['multi_start_workers'] = 0
        else:
            assert trial.status == 'success' and trial.metrics['numerical_parity'] == 1.
            comparison = payload
        assert comparison == anchors.setdefault(case, comparison)
        assert trial.runtime_seconds == trial.metrics['wall_seconds']
        assert abs(sum(trial.metrics[k] for k in ('preparation_seconds', 'point_seconds',
                                                 'plane_seconds')) - trial.runtime_seconds) < 1.e-12
        full_payloads += 1
    if prefix:
        checkout = definition.methods[-1].source_commit
        if prefix == 'pilot-v2.':
            assert all(original.provenance.source_artifacts[k] == final.provenance.source_artifacts[k]
                       for k in SOURCES[:-1])
            tool_path = next(k for k in original.provenance.source_artifacts
                             if k.endswith('tools\\benchmark_registration_preparation.py'))
            final_tool_path = next(k for k in final.provenance.source_artifacts
                                  if k.endswith('tools\\benchmark_registration_preparation.py'))
            assert original.provenance.source_artifacts[tool_path] == final.provenance.source_artifacts[final_tool_path]
            for method in definition.methods[1:]:
                method.source_commit = COMMIT
            definition.limitations.append('Uncommitted tool pilot; source-byte hashes match the final '
                'run and are bound to their later source commit ' + COMMIT + '. Original checkout HEAD: ' + checkout)
        else:
            for method in definition.methods[1:]:
                method.source_commit = None
            definition.limitations.append('Early tool pilot on uncommitted source; original checkout HEAD: '
                + checkout + '. Recorded source-byte hashes are preserved. Two failures are solely the '
                'sequential control recording multi_start_workers=1 in the plane solver options. '
                'All other numerical fields match final cases; raw payloads/statuses/digests stay unchanged. '
                'The final tool passes the same plane options to every method.')
    definition.provenance.source_artifacts.update({
        'original_definition': sha256_path(root / 'registration.definition.json'),
        'original_aggregate': sha256_path(root / 'registration.benchmark.json'),
        **{'published:' + k: v for k, v in bindings.items()},
        **{'committed:' + source: bindings['measured.' + Path(source).name + '.txt'] for source in SOURCES},
    })
    definition.limitations.append('Local path prefixes are normalized to ${WORKSPACE_ROOT} and '
        '${DATASET_ROOT}. Every complete payload and original input/config/output digest is unchanged. '
        'Output digests bind canonical numerical JSON, not a standalone result file.')
    definition_name = f'{prefix}registration.definition.json'
    definition.save(DEST / definition_name)
    aggregate = aggregate_benchmark_definition(definition).model_copy(deep=True)
    # Keep each complete payload in its definition and avoid duplicating large
    # payload strings in the derived summary. Output hashes bind the referent.
    for trial in aggregate.trials:
        trial.provenance.notes[-1] = 'Complete numerical payload retained in ' + definition_name
    aggregate.save(DEST / f'{prefix}registration.benchmark.json')

for number in (1, 2):
    root = Path(f'outputs/public-kitti-registration-preparation-pilot-v{number}')
    helper = Path(f'.tmp-calibrex-registration-pilot{ "" if number == 1 else "-v2" }.py')
    raw = json.loads((root / 'records.json').read_bytes())
    assert len(raw) == 6
    for item in raw:
        case = f'0000000000-0000000115-cap-{item["cap"]}'
        expected = anchors[case]
        assert item['payload']['point'] == expected['point_to_point']
        comparison_plane = json.loads(json.dumps(item['payload']['plane']))
        if number == 1:
            assert 'multi_start_workers' not in comparison_plane['resolved_options']['common_icp']
            comparison_plane['resolved_options']['common_icp']['multi_start_workers'] = 0
        assert comparison_plane == expected['point_to_plane']
        for cloud, bound in zip(item['payload']['points'], expected['prepared']):
            import numpy as np
            assert len(cloud) == bound['count']
            assert digest([p[0] for p in cloud]) == bound['ids_sha256']
            assert hashlib.sha256(np.asarray([p[1] for p in cloud], dtype='<f8').tobytes()).hexdigest() == bound['positions_float64_le_sha256']
        full_payloads += 1
    archived_helper = DEST / f'exploratory-v{number}.launcher.py.txt'
    archived_helper.write_text(normalize(helper.read_text()).rstrip() + '\n',
                               encoding='utf-8', newline='\n')
    wrapped = {
        'provenance': {
            'generator': 'historical exploratory launcher, archived with normalized paths',
            'reference_commit': BASELINE,
            'original_records_sha256': sha256_path(root / 'records.json'),
            'original_launcher_sha256': sha256_path(helper),
            'published_launcher_sha256': sha256_path(archived_helper),
            'raw_inputs': {k: v for k, v in normalize(final.provenance.source_artifacts).items()
                           if k.endswith('.bin') and ('0000000000' in k or '0000000115' in k)},
            'limitations': [
                'Uncommitted exploratory runs; full measured-source byte hashes were not captured. '
                'Do not bind candidate timings to the final source commit or include them in final medians.',
                'Pilot v1 used vectorization sequentially; pilot v2 also used automatic multi-start '
                'workers. No fresh scan read/hash cost was timed; prepared digesting was outside wall.',
                'All solver numerical fields and every bit of prepared positions match the final case. '
                'Pilot v1 predates the multi_start_workers option, so its plane option metadata omits '
                'that field; the final default is 0. Original payloads are unchanged. '
                'All records are retained. This ad-hoc record wrapper is not a standard benchmark schema.',
            ],
        },
        'trials': normalize(raw),
    }
    (DEST / f'exploratory-v{number}.records.txt').write_text(
        json.dumps(wrapped, sort_keys=True, separators=(',', ':')) + '\n',
        encoding='utf-8', newline='\n')

for path in DEST.glob('*.json'):
    assert validate_file(path).valid, path
assert full_payloads == 114
print('Published 6 schema-valid JSON reports and all 114 complete numerical payloads.')
