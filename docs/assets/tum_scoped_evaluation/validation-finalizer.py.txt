"""Bind completed tests and validation to the exact production source capsule."""
import hashlib
import io
import json
from pathlib import Path
import re
import shutil
import subprocess
import tarfile

ROOT=Path(__file__).resolve().parent
TARGET=ROOT/'docs/assets/tum_scoped_evaluation'
VALIDATION=ROOT/'outputs/tse-validation-v1'
RUNTIME='5e5f10522ab792ed9713b2b5c101589156e60b99'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def write(path,value):path.write_bytes((json.dumps(value,indent=2)+'\n').encode())
manifest=json.loads((TARGET/'sources-current-manifest.json').read_bytes())
assert len(manifest)==394
raw=subprocess.check_output(['git','archive','HEAD','src/calibrex'])
with tarfile.open(fileobj=io.BytesIO(raw)) as bundle:
    actual={m.name.removeprefix('src/'):hashlib.sha256(bundle.extractfile(m).read()).hexdigest()
            for m in bundle if m.isfile() and m.name.endswith('.py')}
assert actual==manifest
counts={};logs={}
for mode in ('core','archive'):
    for platform in ('windows','linux'):
        name=f'{mode}-{platform}'
        env=json.loads((VALIDATION/(name+'-environment.json')).read_bytes())
        assert env['exit_status']==0 and env['provenance']['sources']==manifest
        assert env['provenance']['git_commit']==RUNTIME
        assert all(manifest[v['path']]==v['sha256'] for v in env['imports'].values())
        assert all(sha(ROOT/path)==digest for path,digest in env['test_files_sha256'].items())
        log=ROOT/'outputs/tse-core-windows-v1.log' if mode=='core' and platform=='windows' else VALIDATION/(name+'.log')
        text=log.read_text(encoding='utf-8')
        counts[name]=int(re.findall(r'(\d+) passed',text)[-1])
        assert counts[name]==(10 if mode=='archive' else 687 if platform=='windows' else 653)
        if mode=='core' and platform=='linux':assert '1 skipped' in text
        shutil.copyfile(log,TARGET/(name+'.log'))
        shutil.copyfile(VALIDATION/(name+'-environment.json'),TARGET/(name+'-environment.json'))
        logs[name+'.log']=sha(TARGET/(name+'.log'))
for platform in ('linux','win32'):
    p=VALIDATION/f'mypy-{platform}.log'
    assert 'Success: no issues found in 394 source files' in p.read_text()
    shutil.copyfile(p,TARGET/p.name);logs[p.name]=sha(p)
parity=json.loads((VALIDATION/'schema-parity.json').read_bytes())
assert parity['schema_exports_and_ledger']==130 and parity['all_exports_byte_identical']
shutil.copyfile(VALIDATION/'schema-parity.json',TARGET/'schema-parity.json')
for source,name in [
    ('.tmp-calibrex-tse-frozen-tests-v1.py','frozen-test-runner.py.txt'),
    ('.tmp-calibrex-tse-linux-v1.sh','linux-test-runner.sh.txt'),
    ('.tmp-calibrex-tse-schema-parity-v1.py','schema-parity.py.txt'),
    ('tests/unit/test_tum_scoped_evaluation_archive.py','test_tum_scoped_evaluation_archive.py.txt'),
    ('tools/audit_tum_scoped_evaluation.py','auditor.py.txt'),
    ('tools/audit_tum_diagnostic_execution.py','audit-primitives.py.txt'),
]:shutil.copyfile(ROOT/source,TARGET/name)
shutil.copyfile(Path(__file__),TARGET/'validation-finalizer.py.txt')
record={'test_pass_counts':counts,'real_device_cases_windows':34,
        'linux_optional_cuda_module_skipped':1,'mypy_source_modules':{'linux':394,'win32':394},
        'public_schema_exports_and_ledger_byte_identical':130,
        'complete_accepted_production_results_exact':9,
        'provenance':{'runtime_commit':RUNTIME,
                      'source_manifest_sha256':sha(TARGET/'sources-current-manifest.json'),
                      'production_provenance_sha256':sha(TARGET/'provenance.json'),
                      'generator_sha256':sha(Path(__file__)),
                      'typecheck_requirements_sha256':sha(ROOT/'tools/typecheck-requirements.txt')},
        'log_sha256':logs}
for filename,key,marker in (('ruff.log','ruff_passed','All checks passed!'),('mkdocs.log','strict_mkdocs_passed','Documentation built in ')):
    p=VALIDATION/filename
    if p.exists():
        assert marker in p.read_text(encoding='utf-8',errors='replace')
        shutil.copyfile(p,TARGET/filename);record[key]=True;logs[filename]=sha(p)
write(TARGET/'validation.json',record)
index=json.loads((TARGET/'index.json').read_bytes())
index['files']={p.relative_to(TARGET).as_posix():{'sha256':sha(p),'size_bytes':p.stat().st_size}
                for p in sorted(TARGET.rglob('*')) if p.is_file() and p.name!='index.json'}
write(TARGET/'index.json',index)
print('Bound completed source-identical validation:',counts)
