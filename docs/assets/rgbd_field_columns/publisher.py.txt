"""Preserve all measured actual-source field-column workers without rewriting them."""
import gzip
import hashlib
import json
from pathlib import Path
import shutil
import tarfile

ROOT=Path(__file__).resolve().parent
OUT=ROOT/'outputs/rgbd-field-columns-pairs-v1'
TARGET=ROOT/'docs/assets/rgbd_field_columns'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def write(path,value):
    path.write_bytes((json.dumps(value,indent=2,allow_nan=False)+'\n').encode())

def schema(value):
    if value is None:
        return {'type':'null'}
    if isinstance(value,bool):
        return {'type':'boolean'}
    if isinstance(value,int):
        return {'type':'integer'}
    if isinstance(value,float):
        return {'type':'number'}
    if isinstance(value,str):
        return {'type':'string'}
    if isinstance(value,list):
        choices={json.dumps(schema(item),sort_keys=True) for item in value}
        items={} if not choices else json.loads(next(iter(choices))) if len(choices)==1 else {'anyOf':[json.loads(item) for item in sorted(choices)]}
        return {'type':'array','minItems':len(value),'maxItems':len(value),'items':items}
    return {'type':'object','additionalProperties':False,'required':list(value),'properties':{key:schema(item) for key,item in value.items()}}

summary=json.loads((OUT/'summary.json').read_bytes())
assert len(summary['attempts'])==6 and summary['complete_result_and_histories_exact']
TARGET.mkdir(exist_ok=False)
(TARGET/'.gitattributes').write_bytes(b'* -text whitespace=cr-at-eol\n')
for name in ['config.yaml','provenance.json','attempts.json','summary.json','launcher.py.txt']:
    shutil.copyfile(OUT/name,TARGET/name)
for source,name in [('docs/assets/tum_window_processes/inputs.json','inputs.json'),
                    ('tools/benchmark_bspline_lidar_imu.py','freezer.py.txt'),
                    ('tools/benchmark_lazy_runtime_loading.py','identity.py.txt'),
                    ('tools/benchmark_radar_doppler_batches.py','capsule-auditor.py.txt'),
                    ('tools/audit_rgbd_field_columns.py','auditor.py.txt')]:
    shutil.copyfile(ROOT/source,TARGET/name)
shutil.copyfile(Path(__file__),TARGET/'publisher.py.txt')
for variant in ('prior','current'):
    source=OUT/('sources-'+variant)
    shutil.copyfile(source/'manifest.json',TARGET/f'sources-{variant}-manifest.json')
    with tarfile.open(TARGET/f'sources-{variant}.tar.gz','w:gz') as bundle:
        for path in sorted((source/'src').rglob('*.py')):
            bundle.add(path,arcname=path.relative_to(source).as_posix())
records={'measurement':[],'result':[],'traces':[],'provenance':[json.loads((OUT/'provenance.json').read_bytes())],
         'attempts':[json.loads((OUT/'attempts.json').read_bytes())]}
for attempt in summary['attempts']:
    folder=TARGET/attempt['folder']
    folder.mkdir()
    for name in ['measurement.json','result.json.gz','traces.json.gz','stdout.log','stderr.log']:
        shutil.copyfile(OUT/folder.name/name,folder/name)
    for kind,name in [('measurement','measurement.json'),('result','result.json.gz'),('traces','traces.json.gz')]:
        raw=(folder/name).read_bytes()
        records[kind].append(json.loads(gzip.decompress(raw) if name.endswith('.gz') else raw))
schemas={}
for kind,values in records.items():
    choices={json.dumps(schema(value),sort_keys=True) for value in values}
    schemas[kind]=json.loads(next(iter(choices))) if len(choices)==1 else {'anyOf':[json.loads(item) for item in sorted(choices)]}
    schemas[kind]['$schema']='https://json-schema.org/draft/2020-12/schema'
write(TARGET/'record-schemas.json',schemas)
files={p.relative_to(TARGET).as_posix():{'sha256':sha(p),'size_bytes':p.stat().st_size}
       for p in sorted(TARGET.rglob('*')) if p.is_file() and p.name!='index.json'}
write(TARGET/'index.json',{'files':files,'provenance':{'measurement_provenance_sha256':sha(TARGET/'provenance.json'),'publisher_sha256':sha(Path(__file__))}})
print('Preserved',len(files)+1,'original source-bound measurement files')
