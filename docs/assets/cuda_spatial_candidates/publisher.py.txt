from pathlib import Path
import hashlib,json,shutil

source=Path('outputs/cuda-spatial-candidates-run-v1')
root=Path('docs/assets/cuda_spatial_candidates');root.mkdir(exist_ok=False)
record=json.loads((source/'measurement.json').read_bytes())
assert len(record['records'])==18
for name in ('measurement.json','measurement.schema.json','benchmark.py.txt',
             'sorted.cu.txt','heap.cu.txt','sources-manifest.json','sources.tar.gz'):
    shutil.copyfile(source/name,root/name)
shutil.copyfile(__file__,root/'publisher.py.txt')
timings={(r['leafsize'],r['neighbors'],r['method']):r['kernel_median_seconds']
         for r in record['records']}
ratio=timings[16,17,'sorted']/timings[16,17,'heap']
(root/'README.md').write_text(f'''# Public GPU spatial-query candidate measurements

The public GLIM Livox input SHA-256 is
`{record['provenance']['bag_sha256']}`. Measurement source commit:
`{record['provenance']['git_commit']}`. The bound generator captures the sixth
initial odometry registration using the documented dense policy.

All 18 configurations retain exact distance/index parity. At leaf size 16,
the 17-neighbor query median is {1000*timings[16,17,'sorted']:.6f} ms with sorted
insertion and {1000*timings[16,17,'heap']:.6f} ms with a max heap ({ratio:.6f}x).
This is an exploratory warmed query-kernel observation, not a complete
calibration speedup. Runtime adapters remain unchanged.

Five GPU-event batches of ten launches follow the recorded fixed order. Capture,
CPU projection, tree construction, uploads, first compilation, PCA, registration
systems, pose updates, fitting and evaluation are excluded. This uses one
captured registration and does not reserve host load or provide extrinsic truth.

The index binds every timing sample, actual shader, generator, schema, publisher,
imported-source manifest and source archive. Local captured point arrays are
not redistributed; their hash and original bag hash are retained. Reproduce
them with the public bag and the bound generator.

```bash
python -m tools.benchmark_cuda_spatial_candidates --output outputs/query-candidates
python -m tools.audit_cuda_spatial_candidates docs/assets/cuda_spatial_candidates
```

See [scope, interpretation and reproduction](../../benchmarks/cuda_spatial_candidates.md).
''',encoding='utf-8',newline='\n')
files={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(root.iterdir()) if p.is_file()}
(root/'publication-index.json').write_text(json.dumps({
    'schema_version':'slac.cuda_spatial_candidates_publication/v0.1',
    'provenance':{'measurement_sha256':hashlib.sha256((source/'measurement.json').read_bytes()).hexdigest()},
    'files':files,
},indent=2)+'\n',encoding='utf-8',newline='\n')
print({'files':len(files),'heap17speedup':ratio})
