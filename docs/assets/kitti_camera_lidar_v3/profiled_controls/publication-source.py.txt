"""Publish all numerical profile probes, exact frozen sources and schemas."""
import hashlib
import json
import shutil
from datetime import datetime, timezone
from pathlib import Path
from typing import Literal
from pydantic import BaseModel, ConfigDict, Field

ROOT = Path(__file__).resolve().parents[1]
PRIVATE = ROOT / "outputs/camera-lidar-profile-batch-v1"
PUBLIC = ROOT / "docs/assets/kitti_camera_lidar_v3/profiled_controls"
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()


class File(BaseModel):
    model_config = ConfigDict(extra="forbid")
    path: str
    bytes: int = Field(ge=0)
    sha256: str = Field(pattern="^[0-9a-f]{64}$")


class Manifest(BaseModel):
    model_config = ConfigDict(extra="forbid")
    schema_version: Literal["calibrex.profile_control_publication/v0.1"]
    provenance: dict
    files: list[File]


batch = json.loads((PRIVATE / "report.json").read_text())
assert batch["status"] == "complete_diagnostic_only" and not batch["errors"]
assert len(batch["runs"]) == 14
assert batch["provenance"]["generator_sha256"] == sha(ROOT / "outputs/camera-lidar-profile-batch-v1.py")
for name, source in batch["provenance"]["sources"].items():
    assert source["sha256"] == sha(PRIVATE / f"{name}-source.py.txt")
PUBLIC.mkdir(exist_ok=False)
for name in ["report.json", "report.schema.json", *[f"{name}-source.py.txt" for name in batch["provenance"]["sources"]]]:
    shutil.copyfile(PRIVATE / name, PUBLIC / name)
schema = (PRIVATE / batch["runs"][0]["report"]).parent / "report.schema.json"
shutil.copyfile(schema, PUBLIC / "profile.schema.json")
for run in batch["runs"]:
    source = PRIVATE / run["report"]
    assert source.resolve().is_relative_to(PRIVATE.resolve())
    assert sha(source) == run["report_sha256"]
    assert (source.parent / "report.schema.json").read_bytes() == schema.read_bytes()
    target = PUBLIC / run["report"]
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, target)
shutil.copyfile(ROOT / "outputs/camera-lidar-profile-batch-v1.py", PUBLIC / "producer-source.py.txt")
shutil.copyfile(Path(__file__), PUBLIC / "publication-source.py.txt")
(PUBLIC / "manifest.schema.json").write_text(json.dumps(Manifest.model_json_schema(), indent=2), encoding="utf-8")
manifest = Manifest(schema_version="calibrex.profile_control_publication/v0.1", provenance={
    "created_at": datetime.now(timezone.utc).isoformat(), "generator_sha256": sha(Path(__file__)),
    "sensor_payload_published": False,
}, files=[File(path=p.relative_to(PUBLIC).as_posix(), bytes=p.stat().st_size, sha256=sha(p))
          for p in sorted(PUBLIC.rglob("*")) if p.is_file()])
(PUBLIC / "manifest.json").write_text(manifest.model_dump_json(indent=2), encoding="utf-8")
print(len(manifest.files), "files", sum(f.bytes for f in manifest.files), "bytes")
