"""Publish complete audited measurements with only location metadata made portable."""
from __future__ import annotations

import hashlib
import json
import os
import re
import statistics
import subprocess
import sys
from pathlib import Path

workspace = next(p for p in Path(__file__).resolve().parents if (p/'pyproject.toml').is_file())
sys.path.insert(0,str(workspace))
os.chdir(workspace)
from calibrex.core.benchmark import BenchmarkDefinition, BenchmarkArtifact, aggregate_benchmark_definition
from calibrex.core.continuous_time_lidar_artifacts import ContinuousTimeLidarPairArtifact
from calibrex.core.io import read_mapping
from tools import benchmark_adaptive_plane_search as b

baseline = 'eb50c99878706a45eb0e584f3573507313cdc172'
destination = Path('docs/assets/cuda_continuous_time_lidar')
destination.mkdir(parents=True, exist_ok=True)
entries, summaries, rewrite = {}, {}, {}

def sha(data):
    return hashlib.sha256(data).hexdigest()

def save(name, data, **metadata):
    if isinstance(data, dict):
        data=(json.dumps(data,indent=2,allow_nan=False)+'\n').encode()
    (destination/name).parent.mkdir(parents=True,exist_ok=True)
    (destination/name).write_bytes(data)
    entries[name]={'sha256':sha(data),**metadata}

def portable(value):
    if isinstance(value,dict):return {portable(k):portable(v) for k,v in value.items()}
    if isinstance(value,list):return [portable(v) for v in value]
    if isinstance(value,str):
        if value in rewrite:return rewrite[value]
        return value.replace(sys.executable,'<PYTHON>').replace(str(workspace),'<WORKSPACE>').replace(workspace.as_posix(),'<WORKSPACE>').replace('\\','/')
    return value

cohorts=[('full_capture',Path('outputs/public-glim-cuda-continuous-full-v1')),
         ('default_budget',Path('outputs/public-glim-cuda-continuous-default-v1'))]
if Path('outputs/public-glim-cuda-continuous-dense-v1/pipeline.definition.json').is_file():
    cohorts.append(('dense_capture',Path('outputs/public-glim-cuda-continuous-dense-v1')))
for source in b.SOURCES:
    save('measured.'+Path(source).name+'.txt',Path(source).read_bytes(),source=source)
    rewrite[source]='measured.'+Path(source).name+'.txt'
for module in b.COMPILED_MODULES:
    source='src/'+module.replace('.','/')+'.py'
    name='baseline.'+module.rsplit('.',1)[1]+'.py.txt'
    data=subprocess.check_output(['git','show',f'{baseline}:{source}'])
    save(name,data,source=source,source_commit=baseline)
    for _,root in cohorts:
        assert (root/name).read_bytes()==data
        for key in ((root/name).as_posix(),str((root/name).resolve()),(root/name).resolve().as_posix()):
            rewrite[key]=name
for label,root in cohorts:
    definition=BenchmarkDefinition.model_validate(read_mapping(root/'pipeline.definition.json'))
    original_aggregate=BenchmarkArtifact.model_validate(read_mapping(root/'pipeline.benchmark.json'))
    assert aggregate_benchmark_definition(definition)==original_aggregate
    assert len(definition.trials)==10
    candidate=next(m.source_commit for m in definition.methods if m.method_id=='sparse')
    assert next(m.source_commit for m in definition.methods if m.method_id=='baseline')==baseline
    for source in b.SOURCES:
        raw=Path(source).read_bytes()
        assert definition.provenance.source_artifacts[source]==sha(raw)
        assert raw.replace(b'\r\n',b'\n')==subprocess.check_output(['git','show',f'{candidate}:{source}'])
    config=next(k for k in definition.provenance.source_artifacts if k.startswith('examples/'))
    raw=Path(config).read_bytes()
    assert definition.provenance.source_artifacts[config]==sha(raw)
    name='measured.'+Path(config).name+'.txt'
    save(name,raw,source=config,source_commit=candidate)
    rewrite[config]=name
    anchor,split_anchor=None,None
    for trial in definition.trials:
        folder=root/(trial.method_id+'.'+trial.split_id)
        measurement_raw=(folder/'measurement.json').read_bytes()
        measurement=json.loads(measurement_raw)
        saved_raw=(folder/'result.yaml').read_bytes()
        model=ContinuousTimeLidarPairArtifact.model_validate(read_mapping(folder/'result.yaml'))
        canonical=b._canonical(model.model_dump(mode='json',exclude_none=True))
        assert canonical==b._canonical(measurement['result'])==trial.provenance.notes[-1]
        assert sha(saved_raw)==measurement['result_sha256']
        assert sha(canonical.encode())==trial.provenance.output_sha256
        assert trial.status=='success' and trial.metrics['complete_result_matches']==1
        assert trial.runtime_seconds==trial.metrics['subprocess_wall_seconds']
        for key in trial.metrics:
            if key in measurement:assert trial.metrics[key]==measurement[key]
        compared=b._comparison_canonical(measurement['result'])
        if anchor is None:anchor=compared
        assert compared==anchor
        if split_anchor is None:split_anchor=measurement['sampled_split']
        assert measurement['sampled_split']==split_anchor
        metadata=json.loads(trial.provenance.notes[0])
        assert metadata['numba_cache_policy']=='persistent'
        assert metadata['execution_override']==('cuda' if trial.method_id=='sparse' else None)
        if trial.split_id=='repeat-01':
            assert not metadata['numba_cache_before'] and metadata['numba_cache_after']
            assert not metadata['cuda_cache_before']
        if trial.method_id=='sparse':assert metadata['cuda_cache_after']
        else:assert not metadata['cuda_cache_after']
        published=portable(measurement)
        published['result']=measurement['result']
        changed=[k for k in measurement if measurement[k]!=published[k]]
        assert set(changed)<={'loaded_module','loaded_evaluation_modules','loaded_cuda_module'}
        prefix=f'{label}.{folder.name}'
        save(prefix+'.measurement.json',published,original_sha256=sha(measurement_raw),changed_fields=changed)
        save(prefix+'.result.yaml',saved_raw,original_sha256=sha(saved_raw))
    payload=definition.model_dump(mode='json',exclude_none=True)
    for trial in payload['trials']:
        trial['provenance']['notes'][0]=json.dumps(portable(json.loads(trial['provenance']['notes'][0])),sort_keys=True)
        assert portable(trial['provenance']['notes'][-1])==trial['provenance']['notes'][-1]
    published=BenchmarkDefinition.model_validate(portable(payload))
    assert [t.metrics for t in published.trials]==[t.metrics for t in definition.trials]
    assert [t.runtime_seconds for t in published.trials]==[t.runtime_seconds for t in definition.trials]
    save(label+'.definition.json',published.model_dump(mode='json',exclude_none=True),original_sha256=sha((root/'pipeline.definition.json').read_bytes()))
    save(label+'.benchmark.json',aggregate_benchmark_definition(published).model_dump(mode='json',exclude_none=True),original_sha256=sha((root/'pipeline.benchmark.json').read_bytes()))
    medians={m:{k:statistics.median(t.metrics[k] for t in definition.trials if t.method_id==m)
                for k in ['subprocess_wall_seconds','pipeline_seconds','solver_seconds','worker_cpu_seconds']}
             for m in ['baseline','sparse']}
    pairs=[{m:next(t.runtime_seconds for t in definition.trials if t.method_id==m and t.split_id==f'repeat-{i:02d}')
            for m in ['baseline','sparse']} for i in range(1,6)]
    summaries[label]={'candidate_source_commit':candidate,'medians':medians,'pairs':pairs,
        'median_paired_ratio':statistics.median(p['baseline']/p['sparse'] for p in pairs),
        'status':model.status,'train_correspondences':model.train_correspondence_count,
        'holdout_correspondences':model.holdout_correspondence_count,'rank':model.observability.rank,
        'sampled_split':split_anchor}
save('conversion-provenance.json',Path('data/public/glim_versatile/conversion-provenance.json').read_bytes())
diagnostic_root=Path('outputs/public-glim-cuda-continuous-diagnostic-v1')
if (diagnostic_root/'diagnostic.json').exists():
    diagnostic=json.loads((diagnostic_root/'diagnostic.json').read_bytes())
    raw=(diagnostic_root/'worker/measurement.json').read_bytes()
    assert diagnostic['provenance']['measurement_sha256']==sha(raw)
    measurement=json.loads(raw)
    reference=json.loads((cohorts[0][1]/'sparse.repeat-01/measurement.json').read_bytes())
    assert b._canonical(measurement['result'])==b._canonical(reference['result'])
    for source,digest in diagnostic['provenance']['source_sha256'].items():
        path=destination/'diagnostic-launcher.py.txt' if source=='diagnostic-launcher.py.txt' else Path(source)
        assert b._digest(path)==digest
    published=portable(measurement);published['result']=measurement['result']
    save('diagnostic.measurement.json',published,original_sha256=sha(raw))
    save('diagnostic.json',diagnostic)
    save('diagnostic-launcher.py.txt',destination/'diagnostic-launcher.py.txt'.read_bytes())
launcher=Path(__file__).read_text(encoding='utf-8')
start=launcher.index("    if re.search(")
end=launcher.index("    (destination/name).parent.mkdir",start)
launcher=launcher[:start]+launcher[end:]
launcher=launcher.replace("destination/'diagnostic-launcher.py.txt'","destination/'diagnostic-launcher.py.txt'")
launcher=launcher.replace("destination/'diagnostic-launcher.py.txt'.read_bytes()","(destination/'diagnostic-launcher.py.txt').read_bytes()")
save('publication-launcher.py.txt',launcher.encode())
index={'provenance':{'baseline_source_commit':baseline,'generator':'publication-launcher.py.txt',
    'input_sha256':b._verified_inputs(b.REPLAY),'limitations':[
        'Location metadata is portable; original digests and all complete calibration values remain bound.',
        'Single-sensor duplicated-topic execution control, not independent two-sensor accuracy evidence.']},
    'summaries':summaries,'files':entries}
(destination/'publication-index.json').write_text(json.dumps(index,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps(summaries,indent=2),flush=True)
