import hashlib
import json
import sys
from pathlib import Path
import calibrex
from calibrex.evaluation.continuous_time_sliding_window import run_synthetic_sliding_window_recovery
output = Path(sys.argv[1])
output.mkdir(parents=True, exist_ok=True)
report = run_synthetic_sliding_window_recovery(seed=20260820, command=sys.argv)
report.save(output / 'result.json')
root = Path(calibrex.__file__).parent
sources = {str(p.relative_to(root)): hashlib.sha256(p.read_bytes()).hexdigest()
           for p in sorted(root.rglob('*.py'))}
(output / 'source_sha256.json').write_text(json.dumps(sources, indent=2) + '\n')
print(report.policy_status, report.policy_reason)
