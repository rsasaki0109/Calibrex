lock = Lock()
counts = {}

def observe(owner, name):
    native = getattr(owner, name)
    label = (owner.__name__ if hasattr(owner,'__name__') else str(owner))+'.'+name
    def observed(*args, **kwargs):
        start = time.perf_counter()
        try:
            return native(*args, **kwargs)
        finally:
            elapsed = time.perf_counter()-start
            with lock:
                row = counts.setdefault(label, {'calls':0,'sum_seconds':0.0})
                row['calls'] += 1
                row['sum_seconds'] += elapsed
    setattr(owner, name, observed)

for name in ['numba_rig_linearization','numba_rig_residuals','numba_rig_normal_equations',
             '_fused_jacobian_rows','_residual_rows','_normal_rows','_pose_world_rows','_world_rows']:
    observe(a, name)

                   (CudaRigLinearizer,'linearize'),
                   (s,'_factor_for_indices'),(s,'_prepare_source_map')]:
    if hasattr(owner,name): observe(owner,name)
