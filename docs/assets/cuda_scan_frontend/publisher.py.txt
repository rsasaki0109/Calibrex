from pathlib import Path
import hashlib,json,shutil,statistics,tarfile

source=Path('outputs/gpu-front-formal-v1')
root=Path('docs/assets/cuda_scan_frontend');root.mkdir(exist_ok=False)
summary=json.loads((source/'summary.json').read_bytes())
protocol=json.loads((source/'protocol.json').read_bytes())
attempts=json.loads((source/'attempts.json').read_bytes())
assert summary['quality_passed'] and len(attempts)==6
for name in ('protocol.json','attempts.json','summary.json'):
    shutil.copyfile(source/name,root/name)
records={}
for a in attempts:
    folder=root/a['worker'];folder.mkdir()
    for name in ('result.json','measurement.json','worker.log'):
        shutil.copyfile(source/a['worker']/name,folder/name)
    records[a['worker']]=json.loads((folder/'measurement.json').read_bytes())
for variant in ('prior','current'):
    folder=source/f'sources-{variant}'
    shutil.copyfile(folder/'manifest.json',root/f'sources-{variant}-manifest.json')
    with tarfile.open(root/f'sources-{variant}.tar.gz','w:gz') as bundle:
        bundle.add('LICENSE',arcname='LICENSE',recursive=False)
        for p in sorted((folder/'src').rglob('*.py')):
            bundle.add(p,arcname=p.relative_to(folder).as_posix(),recursive=False)
for path,name in [('tools/benchmark_bspline_lidar_imu.py','benchmark.py.txt'),
                  ('tools/schemas/bspline_lidar_imu_comparison.schema.json','comparison.schema.json'),
                  ('tests/unit/test_cuda_scan_preprocessing.py','preprocessing-tests.py.txt'),
                  ('tests/unit/test_cuda_spatial_build.py','spatial-build-tests.py.txt'),
                  (__file__,'publisher.py.txt')]:
    shutil.copyfile(path,root/name)
scopes={scope:{v:statistics.median(records[a['worker']][scope+'_seconds'] for a in attempts
                                  if a['variant']==v) for v in ('prior','current')}
        for scope in ('preprocessing','spatial_build','normal_preparation')}
rows=['| Scope | Previous CUDA median | Prototype CUDA median | Observed ratio |',
      '| --- | ---: | ---: | ---: |']
whole=summary['whole_median_seconds'];fit=summary['fit_median_seconds']
rows.append(f"| Complete worker | {whole['prior']:.3f} s | {whole['current']:.3f} s | {summary['whole_speedup']:.3f}x |")
for scope,title in [('preprocessing','Scan preprocessing'),('spatial_build','Spatial initialization'),
                    ('normal_preparation','Normal preparation')]:
    r=scopes[scope]
    rows.append(f"| {title} | {r['prior']:.3f} s | {r['current']:.3f} s | {r['prior']/r['current']:.3f}x |")
rows.append(f"| Unchanged CPU fits/covariance | {fit['prior']:.3f} s | {fit['current']:.3f} s | {summary['fit_speedup']:.3f}x |")
table='\n'.join(rows)
attempt_table='\n'.join(['| Pair | Method | Whole seconds | CPU fit seconds |',
                         '| --- | --- | ---: | ---: |']+
                        [f"| {a['repeat']} | {a['variant']} | {a['whole_seconds']:.6f} | {records[a['worker']]['fit_seconds']:.6f} |" for a in attempts])
result=json.loads((root/'00-current/result.json').read_bytes())
reports=result['options']['initial_odometry_execution']+result['options']['segment_odometry_execution']
stats={k:sum(r['statistics'].get(k,0) for r in reports) for k in reports[0]['statistics']}
text=f'''# Public GPU scan-front-end prototype comparison

Previous source: `{protocol['baseline_commit']}`. Prototype measurement source:
`{protocol['current_commit']}`. Original public GLIM Livox bag SHA-256:
`{protocol['bag_sha256']}`. Raw data is not redistributed.

{table}

{attempt_table}

All three comparisons have zero rotation, lever-arm, clock, holdout and decision
difference. The audit also checks full per-round fitted covariance, trajectories,
costs and convergence evidence. All final verdicts remain **INCONCLUSIVE** and
this clip provides no independent extrinsic truth.

The first prototype worker records {stats['gpu_preprocess_input_rows']:,} input
rows over {stats['gpu_preprocess_calls']:,} GPU preprocessing calls,
{stats['gpu_preprocess_kernel_calls']:,} crop/centroid kernel calls and no native
preprocessing fallback. It builds {stats['gpu_tree_builds']} device trees using
{stats['gpu_tree_build_kernel_calls']:,} construction kernel calls, and creates
{stats['native_tree_builds']} native trees lazily for uncertainty repairs.

Three alternating fresh-process pairs use one BLAS thread and separate initially
empty CuPy compilation caches. Driver/runtime caches may persist. Host load is
unreserved. Whole times range substantially between attempts; unchanged CPU fits
also vary. These are observed medians, not a precise causal estimate or a
general speed guarantee. The prototype remains a draft because the complete
improvement is below the desired large acceleration.

The same six-second, four-round dense policy uses 0.03 m voxels, three local-map
scans, a 1 nm coordinate grid and returned-motion grids of 1 micrometre
translation and 0.1 microradian rotation-vector components. CPU remains the
default and all grids remain disabled by default. No return, neighbor or
calibration-iteration budget is reduced.

Whole timing includes imports, decoding, GPU setup/compilation, all odometry
passes, association, fitting/covariance, noise updates, holdout, signed controls,
hashes, saving and verified imports. Component observers add no synchronization.
Spatial initialization records constructor wall time; native repair trees created
later belong to registration/normal preparation. Some asynchronous tail work may
be charged to a following scope. Component ratios are not multiplied.

The index binds every complete result, original log, full fitted arrays, exact
source archives/manifests, launcher, schema, test snapshots and publisher.

```bash
python -m tools.audit_bspline_lidar_imu_benchmark docs/assets/cuda_scan_frontend
```

See [implementation, limitations and reproduction](../../benchmarks/cuda_scan_frontend.md).
'''
(root/'README.md').write_text(text,encoding='utf-8',newline='\n')
doc=Path('docs/benchmarks/cuda_scan_frontend.md')
doc.write_text(doc.read_text(encoding='utf-8').replace('<!-- results -->',table),encoding='utf-8',newline='\n')
index={p.relative_to(root).as_posix():hashlib.sha256(p.read_bytes()).hexdigest()
       for p in sorted(root.rglob('*')) if p.is_file()}
(root/'publication-index.json').write_text(json.dumps({
    'schema_version':'slac.bspline_lidar_imu_publication/v0.1',
    'provenance':{'input_protocol_sha256':hashlib.sha256((source/'protocol.json').read_bytes()).hexdigest()},
    'files':index,
},indent=2)+'\n',encoding='utf-8',newline='\n')
print({'bound_files':len(index),'scopes':scopes,'statistics':stats})
